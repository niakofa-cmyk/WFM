🎨 Part 2: 3D GPU Additive Plasma Thruster System (src/game/orbital-native-flight-renderer.ts)

We expand your OrbitalNativeFlightRenderer component with a pre-allocated particle matrix. By updating memory coordinates inside a flat buffer (THREE.BufferAttribute) instead of allocating objects on the fly, the system renders bright engine trails behind pilots and AI escorts with 0% runtime heap overhead.
typescript
// Additions integrated inside artifacts/war-for-mars/src/game/orbital-native-flight-renderer.ts
import * as THREE from 'three';

export class OrbitalNativePlasmaThrusterSystem {
  private geometry: THREE.BufferGeometry;
  private material: THREE.PointsMaterial;
  private points: THREE.Points;
  
  private maxParticles = 3000;
  private positions: Float32Array;
  private dynamics: Float32Array; // Stride per particle: [vx, vy, vz, remainingLife]
  private ringIndex = 0;

  constructor(scene: THREE.Scene) {
    this.positions = new Float32Array(this.maxParticles * 3);
    this.dynamics = new Float32Array(this.maxParticles * 4);

    // Initialize all particles offscreen to completely protect early vertex draws
    for (let i = 0; i < this.maxParticles; i++) {
      this.positions[i * 3 + 1] = -99999.0;
    }

    this.geometry = new THREE.BufferGeometry();
    this.geometry.setAttribute('position', new THREE.BufferAttribute(this.positions, 3));

    // High-intensity color signatures: USSF utilizes neon blue, Accord utilizes warm solar amber
    this.material = new THREE.PointsMaterial({
      size: 4.0,
      vertexColors: true,
      transparent: true,
      opacity: 0.85,
      blending: THREE.AdditiveBlending,
      depthWrite: false
    });

    const colors = new Float32Array(this.maxParticles * 3);
    this.geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));

    this.points = new THREE.Points(this.geometry, this.material);
    scene.add(this.points);
  }

  /**
   * Injects an engine trail point behind a ship's thruster exhaust nozzle.
   */
  public emitPlumeNode(x: number, y: number, z: number, headingRad: number, isUssf: boolean) {
    const idx = this.ringIndex * 3;
    const dynIdx = this.ringIndex * 4;

    const cosH = Math.cos(headingRad);
    const sinH = Math.sin(headingRad);

    // Position particle directly behind the engine emitter location
    this.positions[idx]     = Math.fround(x - sinH * 6.5);
    this.positions[idx + 1] = Math.fround(y);
    this.positions[idx + 2] = Math.fround(z - cosH * 6.5);

    // Eject vectors in the opposite direction of flight momentum paths
    this.dynamics[dynIdx]     = Math.fround(-sinH * 2.0 + (Math.random() - 0.5) * 0.4);
    this.dynamics[dynIdx + 1] = Math.fround((Math.random() - 0.5) * 0.1);
    this.dynamics[dynIdx + 2] = Math.fround(-cosH * 2.0 + (Math.random() - 0.5) * 0.4);
    this.dynamics[dynIdx + 3] = 1.0; // Life scale value

    const colorArray = this.geometry.attributes.color.array as Float32Array;
    if (isUssf) {
      colorArray[idx] = 0.2; colorArray[idx + 1] = 0.5; colorArray[idx + 2] = 1.0; // USSF Blue
    } else {
      colorArray[idx] = 1.0; colorArray[idx + 1] = 0.6; colorArray[idx + 2] = 0.1; // Accord Solar Amber
    }

    this.ringIndex = (this.ringIndex + 1) % this.maxParticles;
  }

  /**
   * Steps particle translation movements across active rendering frame passes.
   */
  public stepThrusterPlumes() {
    const pos = this.geometry.attributes.position.array as Float32Array;

    for (let i = 0; i < this.maxParticles; i++) {
      const idx = i * 3;
      const dynIdx = i * 4;

      if (pos[idx + 1] < -50000.0) continue; // Skip unallocated particle indices

      this.dynamics[dynIdx + 3] = Math.fround(this.dynamics[dynIdx + 3] - 0.04); // Age degradation step

      if (this.dynamics[dynIdx + 3] <= 0.0) {
        pos[idx + 1] = -99999.0; // Expire out of viewport bounds
        continue;
      }

      // Displace position coordinates based on current momentum deltas
      pos[idx]     = Math.fround(pos[idx] + this.dynamics[dynIdx]);
      pos[idx + 1] = Math.fround(pos[idx + 1] + this.dynamics[dynIdx + 1]);
      pos[idx + 2] = Math.fround(pos[idx + 2] + this.dynamics[dynIdx + 2]);
    }

    this.geometry.attributes.position.needsUpdate = true;
  }
}

🎨 Part 3: Integrating Trails into the Shared Renderer

We attach the plasma system directly within the code-split OrbitalNativeFlightRenderer tracking matrix loop:

// Append parameters inside your active artifacts/war-for-mars/src/game/orbital-native-flight-renderer.ts file
import { OrbitalNativePlasmaThrusterSystem } from './orbital-native-plasma-thruster';

export class OrbitalNativeFlightRenderer {
  private thrusterPlumes!: OrbitalNativePlasmaThrusterSystem;
  // ... prior variable mesh layout registries maintain full operational continuity

  private initializeThrusterSystem(scene: THREE.Scene) {
    this.thrusterPlumes = new OrbitalNativePlasmaThrusterSystem(scene);
  }

  public renderUnifiedBattlefield(actors: any[], localZeroX: number, localZeroY: number, currentFaction: string) {
    // ... prior loop geometry translation updates proceed normally
    
    for (let i = 0; i < actors.length; i++) {
      const actor = actors[i];
      const isUssf = actor.factionId === 1;
      const headingRad = actor.heading * (Math.PI / 180);

      // Inject continuous plume nodes relative to current coordinate vectors
      if (this.thrusterPlumes) {
        this.thrusterPlumes.emitPlumeNode(
          actor.x - localZeroX,
          isUssf ? 0.0 : 4.0,
          actor.y - localZeroY,
          headingRad,
          isUssf
        );
      }
    }

    // Step particle physics simulations sequentially before final scene render pass
    if (this.thrusterPlumes) {
      this.thrusterPlumes.stepThrusterPlumes();
    }
  }
}
