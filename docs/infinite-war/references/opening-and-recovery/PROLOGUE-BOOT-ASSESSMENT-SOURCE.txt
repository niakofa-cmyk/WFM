// artifacts/war-for-mars/src/game/wfm-boot.ts
import { ENGINE_CONFIG } from '../../../../lib/deterministic-core';

export enum AppStateNode {
  TITLE_SCREEN = 0,
  HISTORICAL_PROLOGUE = 1,
  MILITARY_HANGAR = 2,
  LIVE_ORBITAL_UNIVERSE = 3
}

export interface PilotProfilePayload {
  pilotId: number;
  credits: number;
  isHullDestroyed: boolean;
  assignedFaction: 'USSF' | 'ACCORD' | 'UNASSIGNED';
  currentSectorX: bigint;
  currentSectorY: bigint;
}

export class WFMBootLifecycleManager {
  /**
   * RECOVERY AUDIT FIX (f84213d): Overrides the unsafe automatic account boot.
   * Forces all profiles through the Hangar configuration lifecycle cleanly.
   */
  public static handleAccountBootSecure(
    authenticatedUser: any,
    currentProfile: PilotProfilePayload
  ): AppStateNode {
    console.log(`[Boot Authorization] Processing state verification gate for Pilot ${currentProfile.pilotId}`);

    // Safeguard 1: Permanent enforcement of hull destruction states across reboots
    if (currentProfile.isHullDestroyed) {
      console.warn(`[Recovery Safeguard] Pilot ${currentProfile.pilotId} re-authenticated with a broken hull asset. Routing to Hangar Repair Bay.`);
      return AppStateNode.MILITARY_HANGAR; // Buried menu bypass safeguard
    }

    // Safeguard 2: Redirect unassigned rookies directly to the cinematic backstory timeline
    if (currentProfile.assignedFaction === 'UNASSIGNED') {
      return AppStateNode.HISTORICAL_PROLOGUE;
    }

    // Default configuration: Safe hangar landing zone zone. No forced combat exposure.
    return AppStateNode.MILITARY_HANGAR;
  }

  /**
   * Enforces strict verification checks when changing ship layouts or rejoining maps.
   * Completely eliminates the health restoration exploit found in your audit.
   */
  public static validateRejoinIntegrity(
    oldState: PilotProfilePayload, 
    requestedModifications: any
  ): boolean {
    if (oldState.isHullDestroyed) {
      console.error(`[Exploit Intercepted] Blocked layout alteration request on a destroyed vehicle grid.`);
      return false; // Blocks invalid state updates from injecting raw health points
    }
    return true;
  }
}

⚛️ Part 2: The Unified Gateway App Frame Controller (src/game/WarApp.tsx)

// artifacts/war-for-mars/src/game/WarApp.tsx
import React, { useState, useEffect } from 'react';
import { AppStateNode, WFMBootLifecycleManager, PilotProfilePayload } from './wfm-boot';
import { HistoricalArchive } from './HistoricalArchive';
import { GalaxyScreen } from './GalaxyScreen';

export const WarApp: React.FC = () => {
  const [appState, setAppState] = useState<AppStateNode>(AppStateNode.TITLE_SCREEN);
  const [pilotProfile, setPilotProfile] = useState<PilotProfilePayload>({
    pilotId: 101,
    credits: 90,
    isHullDestroyed: false,
    assignedFaction: 'UNASSIGNED', // Defaults to unassigned to trigger prologue requirements
    currentSectorX: 0n,
    currentSectorY: 0n
  });

  // Mock initial account authentication check trigger pass
  useEffect(() => {
    const timer = setTimeout(() => {
      // Run the profile database state records through the secure filter gate
      const targetNode = WFMBootLifecycleManager.handleAccountBootSecure(null, pilotProfile);
      setAppState(targetNode);
    }, 1000);
    return () => clearTimeout(timer);
  }, []);

  const handleFactionSelection = (faction: 'USSF' | 'ACCORD') => {
    setPilotProfile(prev => ({ ...prev, assignedFaction: faction }));
    setAppState(AppStateNode.MILITARY_HANGAR);
  };

  const executePrimaryUniverseLaunch = () => {
    // Verify structural loadout restrictions before launching into flight coordinates
    const canLaunch = WFMBootLifecycleManager.validateRejoinIntegrity(pilotProfile, null);
    if (canLaunch) {
      setAppState(AppStateNode.LIVE_ORBITAL_UNIVERSE);
    } else {
      alert("🚨 LAUNCH HALTED: Vessel core damage detected. Initiate repair matrices inside Hangar Bay.");
    }
  };

  return (
    <div className="w-full min-h-screen bg-slate-950 text-slate-100 font-mono select-none overflow-hidden">
      {appState === AppStateNode.TITLE_SCREEN && (
        <div className="w-full min-h-screen flex flex-col items-center justify-center bg-black animate-pulse">
          <div className="text-sm font-bold tracking-widest text-red-500 uppercase">WAR FOR MARS: INFINITE WAR</div>
          <div className="text-[10px] text-slate-500 font-mono mt-2">AUTHORIZING SECURE WORKSPACE PARITY GATE...</div>
        </div>
      )}

      {appState === AppStateNode.HISTORICAL_PROLOGUE && (
        <HistoricalArchive onSkipPrologue={() => handleFactionSelection('USSF')} />
      )}

      {appState === AppStateNode.MILITARY_HANGAR && (
        <div className="w-full min-h-screen flex flex-col items-center justify-center p-8 bg-slate-900/20">
          <div className="w-full max-w-2xl border border-slate-800 rounded-xl p-6 bg-slate-950/90 shadow-2xl space-y-4">
            <div className="flex justify-between items-center border-b border-slate-900 pb-3">
              <div>🛰️ MILITARY COMMAND HANGAR [PILOT HUB]</div>
              <div className="text-cyan-400 font-bold text-xs">{pilotProfile.credits} CREDITS</div>
            </div>
            <p className="text-xs text-slate-400 leading-relaxed">
              Vessel tracking indices are nominal. Systems are secured within the safe command perimeter zone. 
              Configure module configurations and select deployment coordinates when ready.
            </p>
            <div className="bg-slate-900/60 p-3 rounded border border-slate-800 text-[11px] text-slate-400">
              STATUS CHANNELS: <span className="text-emerald-400 font-bold">MULTIPLAYER REPLICATIONS STANDBY</span>
            </div>
            <button
              onClick={executePrimaryUniverseLaunch}
              className="w-full py-3 bg-gradient-to-r from-red-950 to-red-900 hover:from-red-900 hover:to-red-800 text-red-400 font-bold text-xs uppercase tracking-wider rounded border border-red-700/40 shadow-lg shadow-red-950/20"
            >
              🚀 Launch Active Fighter Into Persistent Universe Gateway
            </button>
          </div>
        </div>
      )}

      {appState === AppStateNode.LIVE_ORBITAL_UNIVERSE && (
        <div className="relative">
          <GalaxyScreen />
          {/* Operational Menu Disconnect / Return Link Overlay */}
          <button
            onClick={() => setAppState(AppStateNode.MILITARY_HANGAR)}
            className="absolute top-4 right-4 bg-slate-950/90 border border-slate-800 hover:border-slate-700 px-3 py-1.5 rounded font-mono text-[10px] text-slate-400 font-bold z-50 transition-all uppercase"
          >
            🔌 Disengage Drive (Return to Safe Hangar)
          </button>
        </div>
      )}
    </div>
  );
};
