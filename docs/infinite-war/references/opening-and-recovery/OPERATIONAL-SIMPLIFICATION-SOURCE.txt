// artifacts/war-for-mars/src/game/Orbital3DCanvas.tsx
import React, { useEffect, useRef } from 'react';
import * as THREE from 'three';
import { SharedSectorPayload } from './types';

export const Orbital3DCanvas: React.FC<{ payload: SharedSectorPayload }> = ({ payload }) => {
  const mountRef = useRef<HTMLDivElement | null>(null);
  const rendererRef = useRef<THREE.WebGLRenderer | null>(null);
  const ussfMeshRef = useRef<THREE.InstancedMesh | null>(null);
  const accordMeshRef = useRef<THREE.InstancedMesh | null>(null);

  useEffect(() => {
    if (!mountRef.current) return;

    const scene = new THREE.Scene();
    scene.background = new THREE.Color('#010105');

    const camera = new THREE.PerspectiveCamera(60, 800 / 500, 0.1, 30000);
    const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
    rendererRef.current = renderer;
    renderer.setSize(800, 500);
    mountRef.current.appendChild(renderer.domElement);

    // Initialize High-Performance Faction Ships Instancing Layer
    const ussfGeom = new THREE.ConeGeometry(3, 11, 4);
    ussfGeom.rotateX(Math.PI / 2);
    const ussfMat = new THREE.MeshBasicMaterial({ color: '#3b82f6', wireframe: true });
    const ussfMesh = new THREE.InstancedMesh(ussfGeom, ussfMat, 500);
    ussfMeshRef.current = ussfMesh;
    scene.add(ussfMesh);

    const accordGeom = new THREE.BoxGeometry(5, 2, 10);
    const accordMat = new THREE.MeshBasicMaterial({ color: '#f59e0b', wireframe: true });
    const accordMesh = new THREE.InstancedMesh(accordGeom, accordMat, 500);
    accordMeshRef.current = accordMesh;
    scene.add(accordMesh);

    camera.position.set(0, 480, 120);
    camera.lookAt(0, 0, 0);

    let frameId: number;
    const animate = () => {
      renderer.render(scene, camera);
      frameId = requestAnimationFrame(animate);
    };
    animate();

    return () => {
      cancelAnimationFrame(frameId);
      ussfGeom.dispose();
      ussfMat.dispose();
      accordGeom.dispose();
      accordMat.dispose();
      renderer.dispose();
      if (mountRef.current && renderer.domElement) {
        mountRef.current.removeChild(renderer.domElement);
      }
    };
  }, []);

  // Structural State Interpolation Bridge (Bypasses standard React updates)
  useEffect(() => {
    const uMesh = ussfMeshRef.current;
    const aMesh = accordMeshRef.current;
    if (!uMesh || !aMesh) return;

    const dummy = new THREE.Object3D();
    let uIdx = 0;
    let aIdx = 0;

    payload.activeActors.forEach((actor) => {
      // Calculate continuous offset tracking values cleanly relative to local zeroes
      dummy.position.set(actor.x - payload.localZeroX, 0, actor.y - payload.localZeroY);
      dummy.rotation.y = actor.heading * (Math.PI / 180);
      dummy.updateMatrix();

      if (actor.factionId === 1 && uIdx < 500) {
        uMesh.setMatrixAt(uIdx++, dummy.matrix);
      } else if (actor.factionId === 2 && aIdx < 500) {
        aMesh.setMatrixAt(aIdx++, dummy.matrix);
      }
    });

    // Clear remainder indices out of field-of-view matrices
    while (uIdx < 500) { dummy.position.set(0, -99999, 0); dummy.updateMatrix(); uMesh.setMatrixAt(uIdx++, dummy.matrix); }
    while (aIdx < 500) { dummy.position.set(0, -99999, 0); dummy.updateMatrix(); aMesh.setMatrixAt(aIdx++, dummy.matrix); }

    uMesh.instanceMatrix.needsUpdate = true;
    aMesh.instanceMatrix.needsUpdate = true;
  }, [payload]);

  return <div ref={mountRef} className="w-full h-[500px]" />;
};
export default Orbital3DCanvas;

📡 Part 2: Fleet-Size-Dependent Encounter Radius Trigger Engine (artifacts/api-server/src/game/encounter-trigger-engine.ts)

This authoritative server component replaces all individual client detection loops. It calculates dynamic proximity vectors, scaling sensory footprint layers relative to the formation size and composition of the fleet. Large battlegroups throw off vast electronic signatures, spawning combat zones fields away, while stealth recon groups require deep proximity.
typescript
// artifacts/api-server/src/game/encounter-trigger-engine.ts
import { ENGINE_CONFIG, createDeterministicVector, Vector2D } from '../../../../lib/deterministic-core';

export interface AutomatedBattleShard {
  encounterId: string;
  fleetId: string;
  activePilotIds: number[];
  threatScale: 'ALPHA' | 'BRAVO' | 'CRITICAL_OMEGA';
  globalLocation: Vector2D;
}

export class EncounterTriggerEngine {
  private trackingEncounters: Map<string, AutomatedBattleShard> = new Map();

  /**
   * Computes authoritative signature footprint bounds based on fleet sizes.
   * Large capital fleet blocks widen the physical engagement zone drastically.
   */
  public getAuthoritativeSensoryRadius(formationSize: number): number {
    const baseProximityBounds = 800.0; // 800m minimum for 2-ship scout units
    // Expand radar signature boundaries by 400m for each additional fleet vessel asset
    return Math.fround(baseProximityBounds + (formationSize - 2) * 400.0);
  }

  /**
   * Scans grid locations across pilots and active patrols to trigger shared combat shards.
   */
  public scanSectorProximityTriggers(
    connectedPilots: Map<number, { factionId: number; x: number; y: number; id: number }>,
    patrolFleets: any[],
    onCombatSpawned: (shard: AutomatedBattleShard) => void
  ): void {
    patrolFleets.forEach((fleet) => {
      const activeSensoryThreshold = this.getAuthoritativeSensoryRadius(fleet.formationSize);

      connectedPilots.forEach((pilot) => {
        // Prevent allied assets from firing friendly trigger checks
        const pilotFactionCode = pilot.factionId; // 1 = USSF, 2 = ACCORD
        const fleetFactionCode = fleet.faction === 'USSF' ? 1 : 2;
        if (pilotFactionCode === fleetFactionCode) return;

        const dx = Math.fround(pilot.x - fleet.globalX);
        const dy = Math.fround(pilot.y - fleet.globalY);
        const distance = Math.fround(Math.sqrt(Math.fround(dx * dx + dy * dy)));

        const encounterId = `shard-combat-${fleet.fleetId}`;

        // 1. Proximity Trigger: Pilot enters fleet's scalable sensory footprint zone
        if (distance <= activeSensoryThreshold) {
          if (!this.trackingEncounters.has(encounterId)) {
            const threatScale = fleet.formationSize >= 4 ? 'CRITICAL_OMEGA' : fleet.formationSize === 3 ? 'BRAVO' : 'ALPHA';

            const newShard: AutomatedBattleShard = {
              encounterId,
              fleetId: fleet.fleetId,
              activePilotIds: [pilot.id],
              threatScale,
              globalLocation: createDeterministicVector(fleet.globalX, fleet.globalY)
            };

            this.trackingEncounters.set(encounterId, newShard);
            onCombatSpawned(newShard);
            console.log(`[Authoritative Engagement] Threat ${threatScale} launched. Fleet ${fleet.fleetId} intercepted Pilot ${pilot.id} at ${distance.toFixed(1)}m.`);
          } else {
            // NATIVE MULTIPLAYER ASSISTANCE: Seamlessly slide an approaching wingman into the identical session
            const ongoingShard = this.trackingEncounters.get(encounterId)!;
            if (!ongoingShard.activePilotIds.includes(pilot.id)) {
              ongoingShard.activePilotIds.push(pilot.id);
              this.trackingEncounters.set(encounterId, ongoingShard);
              console.log(`[Multiplayer Co-Op Assist] Pilot ${pilot.id} joined ongoing battle shard against fleet: ${fleet.fleetId}`);
            }
          }
        } else {
          // 2. Clear engagement state if all participants escape the footprint envelope bounds
          if (this.trackingEncounters.has(encounterId)) {
            const ongoingShard = this.trackingEncounters.get(encounterId)!;
            ongoingShard.activePilotIds = ongoingShard.activePilotIds.filter(id => id !== pilot.id);

            if (ongoingShard.activePilotIds.length === 0) {
              this.trackingEncounters.delete(encounterId);
              console.log(`[Combat Resolved] Fleet ${fleet.fleetId} signature tracking lost. Shard closed.`);
            }
          }
        }
      });
    });
  }
}

// artifacts/war-for-mars/src/game/orbital-radar-sweep.ts
import * as THREE from 'three';

export class OrbitalRadarSweepSystem {
  private sweepMesh: THREE.Mesh;
  private material: THREE.MeshBasicMaterial;
  private currentScale = 0.0;
  private maxRadius = 1.0;
  private isActive = false;

  constructor(scene: THREE.Scene) {
    const geometry = new THREE.RingGeometry(0.96, 1.0, 32);
    geometry.rotateX(-Math.PI / 2);

    this.material = new THREE.MeshBasicMaterial({
      color: '#ef4444',
      transparent: true,
      opacity: 0.0,
      blending: THREE.AdditiveBlending,
      depthWrite: false
    });

    this.sweepMesh = new THREE.Mesh(geometry, this.material);
    this.sweepMesh.position.set(0, -99999, 0);
    scene.add(this.sweepMesh);
  }

  public triggerSweepPulse(x: number, z: number, sensoryRadius: number) {
    this.maxRadius = sensoryRadius;
    this.sweepMesh.position.set(x, 0.5, z);
    this.isActive = true;
  }

  public stepRadarAnimation() {
    if (!this.isActive) return;

    this.currentScale += 0.015;
    if (this.currentScale >= 1.0) this.currentScale = 0.0;

    const currentRadius = this.maxRadius * this.currentScale;
    this.sweepMesh.scale.set(currentRadius, 1.0, currentRadius);
    this.material.opacity = Math.fround(1.0 - this.currentScale) * 0.35;
  }
}

2. Server-Side Co-Op Assist Tracker & Reward Splitter (lib/coop-reward-splitter.ts)

This class updates player logs on the server. It detects all pilot IDs involved in a sector battle shard, allocating credits and tally stats evenly across team participants.
typescript
// lib/coop-reward-splitter.ts
import { AutomatedBattleShard } from '../artifacts/api-server/src/game/encounter-trigger-engine';

export interface PayoutReceipt {
  pilotId: number;
  creditsEarned: number;
  newTotalCredits: number;
}

export class CoopRewardSplitter {
  /**
   * Distributes credits split evenly across all pilots tracked inside the combat shard.
   * Gives an additional 15% co-op assistance credit bonus for each extra wingman involved.
   */
  public static distributeSharedBounty(
    shard: AutomatedBattleShard,
    baseBountyValue: number,
    dbProfiles: Map<number, { credits: number; kills: number }>
  ): PayoutReceipt[] {
    const headcount = shard.activePilotIds.length;
    if (headcount === 0) return [];

    // Scale pooled rewards based on multiplayer co-op assistance metrics
    const assistanceBonus = Math.fround(1.0 + (headcount - 1) * 0.15);
    const totalPooledCredits = Math.floor(Math.fround(baseBountyValue * assistanceBonus));
    const evenlySplitPayout = Math.floor(totalPooledCredits / headcount);

    const receipts: PayoutReceipt[] = [];

    shard.activePilotIds.forEach((pilotId) => {
      const profile = dbProfiles.get(pilotId) || { credits: 90, kills: 14 };
      
      profile.credits += evenlySplitPayout;
      profile.kills += 1; // Secure confirmed kill status totals safely
      dbProfiles.set(pilotId, profile);

      receipts.push({
        pilotId,
        creditsEarned: evenlySplitPayout,
        newTotalCredits: profile.credits
      });
    });

    return receipts;
  }
}

// artifacts/war-for-mars/src/game/BattleChatterConsole.tsx
import React, { useState, useEffect } from 'react';

export const BattleChatterConsole: React.FC<{ activeEncounter: any }> = ({ activeEncounter }) => {
  const [messages, setMessages] = useState<string[]>([]);
  
  const radioComms = [
    "USSF Wingman: 'Form up on my vector! Formations tight!'",
    "Accord Fleet: 'Scout network compromised. Mobilizing capital interceptor matrices.'",
    "USSF Wingman: 'Main shields taking heavy structural friction! Requesting backup loops!'",
    "Accord Fleet: 'Target vectors locked. Discharging auxiliary pulse cannons.'"
  ];

  useEffect(() => {
    if (!activeEncounter?.isCombatEngaged) {
      setMessages([]);
      return;
    }

    setMessages(["[SYSTEM] Tactical battle grid linked. Secure comms operational."]);

    const chatterInterval = setInterval(() => {
      const rollIdx = Math.floor(Math.random() * radioComms.length);
      setMessages(prev => [radioComms[rollIdx], ...prev.slice(0, 3)]);
    }, 4000);

    return () => clearInterval(chatterInterval);
  }, [activeEncounter]);

  if (messages.length === 0) return null;

  return (
    <div className="w-full max-w-4xl bg-slate-950/90 border border-slate-900 rounded-xl p-4 mt-4 font-mono text-[11px] text-slate-300 shadow-2xl h-36 overflow-y-auto box-border">
      <div className="text-[10px] uppercase font-bold text-slate-500 mb-2 border-b border-slate-900 pb-1">🛰️ Real-Time Sector Fleet Comms Log Channel</div>
      {messages.map((msg, i) => <div key={i} className="py-0.5 animate-fadeIn">X_TRANS_LOG // {msg}</div>)}
    </div>
  );
};
