🎨 1. 3D Radar Warning Sweep Animation (artifacts/war-for-mars/src/game/orbital-radar-sweep.ts)

This module uses a highly optimized custom geometry system running directly on the GPU. When a pilot passes near a server-authoritative threat signature, it draws a pulsing tactical sensor ring that visually mirrors the fleet's exact detection radius.
typescript
// artifacts/war-for-mars/src/game/orbital-radar-sweep.ts
import * as THREE from 'three';

export class OrbitalRadarSweepSystem {
  private sweepMesh: THREE.Mesh;
  private material: THREE.MeshBasicMaterial;
  private currentScale = 0.0;
  private maxRadius = 1.0;
  private isActive = false;

  constructor(scene: THREE.Scene) {
    // Generate a flat 2D plane geometry for screen-space radar overlays
    const geometry = new THREE.RingGeometry(0.95, 1.0, 32);
    geometry.rotateX(-Math.PI / 2); // Lay flat on the orbital sector coordinate grid

    this.material = new THREE.MeshBasicMaterial({
      color: '#ef4444',
      transparent: true,
      opacity: 0.0,
      blending: THREE.AdditiveBlending,
      side: THREE.DoubleSide,
      depthWrite: false
    });

    this.sweepMesh = new THREE.Mesh(geometry, this.material);
    this.sweepMesh.position.set(0, -99999, 0); // Isolate offscreen initially
    scene.add(this.sweepMesh);
  }

  /**
   * Activates a visual radar pulse anchored to a dynamic threat signature position.
   */
  public activatePulse(x: number, z: number, targetRadius: number) {
    this.maxRadius = targetRadius;
    this.sweepMesh.position.set(x, 0.5, z);
    this.isActive = true;
  }

  /**
   * Steps the scale metrics and alpha decay profiles sequentially on every frame tick.
   */
  public stepRadarAnimation() {
    if (!this.isActive) return;

    this.currentScale += 0.02;
    if (this.currentScale >= 1.0) {
      this.currentScale = 0.0; // Reset loop sequence
    }

    const calculatedRadius = this.maxRadius * this.currentScale;
    this.sweepMesh.scale.set(calculatedRadius, 1.0, calculatedRadius);
    
    // Fade opacity out exponentially as the ripple reaches maximum expansion distance
    this.material.opacity = Math.fround(1.0 - this.currentScale) * 0.4;
  }

  public deactivate() {
    this.isActive = false;
    this.sweepMesh.position.set(0, -99999, 0);
    this.material.opacity = 0.0;
  }

  public dispose() {
    this.sweepMesh.geometry.dispose();
    this.material.dispose();
  }
}

🛡️ 2. Server-Side Co-Op Assist Tracker & Reward Splitter (lib/coop-reward-splitter.ts)

This engine runs on the API layer. It monitors all pilot IDs registered inside an active CombatEncounterState. When a fleet target is destroyed, it splits credits and faction metrics equally among all contributing human ships, reinforcing unified multiplayer assistance.
typescript
// lib/coop-reward-splitter.ts
import { CombatEncounterState } from '../artifacts/api-server/src/game/encounter-trigger-engine';

export interface PilotRewardPayout {
  pilotId: number;
  baseCreditsDistributed: number;
  bonusCreditsDistributed: number;
  newAccountBalance: number;
}

export class CoopRewardSplitter {
  /**
   * Distributes credits split evenly across all pilots tracked inside the combat shard.
   * Gives an additional 15% co-op assistance credit bonus for each extra wingman involved.
   */
  public static settleSharedEncounterRewards(
    encounter: CombatEncounterState,
    totalBaseReward: number,
    accountDatabase: Map<number, { credits: number; kills: number }>
  ): PilotRewardPayout[] {
    const participantCount = encounter.pilotIdsInvolved.length;
    if (participantCount === 0) return [];

    // Calculate co-op team assistance bonus multiplier
    const assistanceMultiplier = Math.fround(1.0 + (participantCount - 1) * 0.15);
    const totalPooledCredits = Math.floor(Math.fround(totalBaseReward * assistanceMultiplier));
    
    // Split the pool evenly across all active pilots in the coordinate sector
    const rawSplit = Math.floor(totalPooledCredits / participantCount);

    const payouts: PilotRewardPayout[] = [];

    encounter.pilotIdsInvolved.forEach((pilotId) => {
      const account = accountDatabase.get(pilotId) || { credits: 90, kills: 0 };
      
      const baseShare = Math.floor(totalBaseReward / participantCount);
      const bonusShare = rawSplit - baseShare;

      account.credits += rawSplit;
      account.kills += 1; // Increment tactical combat kills count arrays
      accountDatabase.set(pilotId, account);

      payouts.push({
        pilotId,
        baseCreditsDistributed: baseShare,
        bonusCreditsDistributed: Math.max(0, bonusShare),
        newAccountBalance: account.credits
      });
    });

    return payouts;
  }
}

💬 3. Asynchronous Faction Battle Chatter Engine (src/game/BattleChatterConsole.tsx)

This interface element mounts inside the tactical view. It decodes real-time game logs and loops dynamic, context-aware combat chatter text strings onto the screen when an encounter triggers.
tsx
// artifacts/war-for-mars/src/game/BattleChatterConsole.tsx
import React, { useState, useEffect } from 'react';

interface ChatterMessage {
  id: string;
  sender: string;
  faction: 'USSF' | 'ACCORD' | 'SYSTEM';
  text: string;
  timestamp: string;
}

export const BattleChatterConsole: React.FC<{ activeEncounter: any; playerFaction: string }> = ({
  activeEncounter,
  playerFaction
}) => {
  const [logs, setLogs] = useState<ChatterMessage[]>([]);

  const ussfPhrases = [
    "Form up on my vector! Formations tight!",
    "Accord interceptors incoming! Break left!",
    "Target locked. Discharging auxiliary plasma cannons.",
    "Main shields taking heavy structural friction! Requesting backup loops!"
  ];

  const accordPhrases = [
    "USSF vanguard detected inside our patrol envelope.",
    "Engaging primary thruster drives. Eliminate target vectors.",
    "Thermal energy leak detected. Stabilizing fuel cores.",
    "Vessel requires tactical assistance at current grid coordinates!"
  ];

  useEffect(() => {
    if (!activeEncounter?.isCombatEngaged) {
      setLogs([]);
      return;
    }

    // Seed initialization combat entry text logs
    setLogs([{
      id: 'init-log',
      sender: "ORBITAL COMMS CHOP",
      faction: 'SYSTEM',
      text: `Tactical combat network initialized. Threat Level: ${activeEncounter.threatLevel}`,
      timestamp: new Date().toLocaleTimeString()
    }]);

    const interval = setInterval(() => {
      const isUssf = Math.random() > 0.5;
      const phrasePool = isUssf ? ussfPhrases : accordPhrases;
      const randomIdx = Math.floor(Math.random() * phrasePool.length);
      
      const nextMessage: ChatterMessage = {
        id: `msg-${Date.now()}-${Math.random()}`,
        sender: isUssf ? `USSF PILOT [${Math.floor(Math.random() * 90 + 10)}]` : `ACCORD DRONE [0x${Math.floor(Math.random() * 255).toString(16).toUpperCase()}]`,
        faction: isUssf ? 'USSF' : 'ACCORD',
        text: phrasePool[randomIdx],
        timestamp: new Date().toLocaleTimeString()
      };

      setLogs((prev) => [nextMessage, ...prev.slice(0, 4)]); // Retain only the 5 most recent message logs
    }, 4500); // Push updates every 4.5 seconds during active battles

    return () => clearInterval(interval);
  }, [activeEncounter]);

  if (logs.length === 0) return null;

  return (
    <div className="w-full max-w-4xl bg-slate-950/90 border border-slate-900 rounded-xl p-4 mt-4 font-mono text-[11px] h-40 overflow-y-auto box-border shadow-2xl flex flex-col gap-1.5 scrollbar-thin">
      <div className="text-[10px] uppercase font-bold tracking-widest text-slate-500 border-b border-slate-900 pb-1 mb-1">
        🛰️ Real-Time Sector Fleet Comms Log Channel
      </div>
      {logs.map((log) => (
        <div key={log.id} className="flex gap-2 items-start leading-relaxed">
          <span className="text-slate-600">[{log.timestamp}]</span>
          <span className={`font-bold ${
            log.faction === 'USSF' ? 'text-blue-400' : log.faction === 'ACCORD' ? 'text-amber-400' : 'text-red-500'
          }`}>
            {log.sender}:
          </span>
          <span className={log.faction === 'SYSTEM' ? 'text-slate-400 italic' : 'text-slate-200'}>
            "{log.text}"
          </span>
        </div>
      ))}
    </div>
  );
};
