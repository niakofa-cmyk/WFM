🛡️ Part 1: Fleet-Size-Dependent Encounter Radius Trigger Engine (artifacts/api-server/src/game/encounter-trigger-engine.ts)

Instead of utilizing static client-side radar bubbles, this engine computes dynamic sensor envelopes on the server. The detection and combat lock radii scale based on the size and composition of the approaching fleet. A massive battleship squadron emits a huge signature that can be detected from sectors away, while an agile scouting duo requires tight proximity to trigger an ensuing battle.
typescript
// artifacts/api-server/src/game/encounter-trigger-engine.ts
import { ENGINE_CONFIG, createDeterministicVector, Vector2D } from '../../../../lib/deterministic-core';
import { FactionFleet } from './universe-patrols';
import { AuthorizedPilotState } from './shared-flight-session';

export interface CombatEncounterState {
  encounterId: string;
  fleetId: string;
  pilotIdsInvolved: number[];
  originSectorX: bigint;
  originSectorY: bigint;
  isCombatEngaged: boolean;
  threatLevel: 'ALPHA' | 'BRAVO' | 'OMEGA';
}

export class EncounterTriggerEngine {
  private activeEncounters: Map<string, CombatEncounterState> = new Map();

  /**
   * Dynamically calculates a fleet's sensor signature footprint.
   * Larger fleet sizes drastically increase the radius of the engagement trigger zone.
   */
  public calculateFleetSensoryRadius(formationSize: number): number {
    const baseRadius = 800.0; // 800 meters minimum for scouts (2 ships)
    // Scale radius by 400m per additional ship in formation to model thermal signatures
    return Math.fround(baseRadius + (formationSize - 2) * 400.0);
  }

  /**
   * Scans active server pilot vectors against roaming faction fleets.
   * If a proximity threshold is crossed, a shared combat encounter triggers on the server.
   */
  public evaluateGlobalProximityTriggers(
    livePilots: Map<number, AuthorizedPilotState>,
    roamingFleets: FactionFleet[],
    onEncounterSpawned: (encounter: CombatEncounterState) => void
  ): void {
    roamingFleets.forEach((fleet) => {
      const fleetSensoryRadius = this.calculateFleetSensoryRadius(fleet.formationSize);
      
      livePilots.forEach((pilot) => {
        // Prevent allied units from triggering combat loops against their own faction
        if (pilot.faction === fleet.faction) return;

        const dx = Math.fround(pilot.position.x - fleet.globalX);
        const dy = Math.fround(pilot.position.y - fleet.globalY);
        const distance = Math.fround(Math.sqrt(Math.fround(dx * dx + dy * dy)));

        const encounterKey = `enc-${fleet.fleetId}`;

        // 1. Proximity Check: Pilot enters fleet's dynamic combat signature zone
        if (distance <= fleetSensoryRadius) {
          if (!this.activeEncounters.has(encounterKey)) {
            const threatLevel = fleet.formationSize >= 4 ? 'OMEGA' : fleet.formationSize === 3 ? 'BRAVO' : 'ALPHA';
            
            const newEncounter: CombatEncounterState = {
              encounterId: encounterKey,
              fleetId: fleet.fleetId,
              pilotIdsInvolved: [pilot.pilotId],
              originSectorX: BigInt(Math.floor(fleet.globalX / ENGINE_CONFIG.CHUNK_SIZE)),
              originSectorY: BigInt(Math.floor(fleet.globalY / ENGINE_CONFIG.CHUNK_SIZE)),
              isCombatEngaged: true,
              threatLevel
            };

            this.activeEncounters.set(encounterKey, newEncounter);
            onEncounterSpawned(newEncounter);
            console.log(`[Authoritative Engagement] Threat ${threatLevel} spawned. Fleet ${fleet.fleetId} intercepted Pilot ${pilot.pilotId} at range ${distance.toFixed(1)}m.`);
          } else {
            // Native Multi-Player Assistance: Append allied human pilots seamlessly to ongoing battles
            const ongoingCombat = this.activeEncounters.get(encounterKey)!;
            if (!ongoingCombat.pilotIdsInvolved.includes(pilot.pilotId)) {
              ongoingCombat.pilotIdsInvolved.push(pilot.pilotId);
              this.activeEncounters.set(encounterKey, ongoingCombat);
              console.log(`[Multiplayer Assist Link] Pilot ${pilot.pilotId} dropped directly into Battle Shard: ${encounterKey}`);
            }
          }
        } else {
          // 2. Clear engagement state if all pilots retreat beyond outer signature boundaries
          if (this.activeEncounters.has(encounterKey)) {
            const ongoingCombat = this.activeEncounters.get(encounterKey)!;
            ongoingCombat.pilotIdsInvolved = ongoingCombat.pilotIdsInvolved.filter(id => id !== pilot.pilotId);
            
            if (ongoingCombat.pilotIdsInvolved.length === 0) {
              this.activeEncounters.delete(encounterKey);
              console.log(`[Combat Dissolved] Fleet ${fleet.fleetId} lost visual tracking. Engagement cleared.`);
            }
          }
        }
      });
    });
  }
}

import { EncounterTriggerEngine } from '../game/encounter-trigger-engine';
import { TelemetrySecurityGate } from '../../../../lib/security-gate';

export const worldOpsRouter = Router();
const encounterEngine = new EncounterTriggerEngine();

// Cleaned Server Route: Aggregates multiplayer ticks without instanced mission overhead
worldOpsRouter.post('/process-universe-frame', async (req: Request, res: Response): Promise<void> => {
  try {
    const { payload, signature } = req.body;
    
    // 1. Sanitize incoming telemetry payload array stream
    const isSecure = TelemetrySecurityGate.verifyPayloadSanity(payload, signature, "TRANS_SECRET_HANDSHAKE_KEY_TOKEN_73921");
    if (!isSecure) {
      res.status(403).json({ error: "HTTP 403: Security authentication rejected framework trace flags." });
      return;
    }

    // 2. (Physics execution step processes positions via shared flight session states)
    // Dynamic engagement state evaluation runs inline inside server loops

      }

      success: true,
      serverTick: payload.tick,
      activeEncounterAlert: structuralEncounterAlert,
      msg: "Unified open universe telemetry synced cleanly."
    });

  } catch (error) {
