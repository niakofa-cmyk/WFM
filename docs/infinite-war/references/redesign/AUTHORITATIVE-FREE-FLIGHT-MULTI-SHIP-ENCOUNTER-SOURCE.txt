1. Server-Side Universe Patrol & Fleet Movement Engine (artifacts/api-server/src/game/universe-patrols.ts)

This class runs on your Node.js API layer. It maintains the persistent coordinate vectors for both the USSF (United States Space Force) and Accord fleets, stepping their movements every two minutes across world coordinate lines.

// artifacts/api-server/src/game/universe-patrols.ts
import { ENGINE_CONFIG, createDeterministicVector, Vector2D } from '../../../../lib/deterministic-core';

export interface FactionFleet {
  fleetId: string;
  faction: 'USSF' | 'ACCORD';
  globalX: number;
  globalY: number;
  formationSize: number;
  currentVelocity: Vector2D;
  targetSectorX: number;
  targetSectorY: number;
}

export class UniversePatrolEngine {
  private fleets: Map<string, FactionFleet> = new Map();

  constructor() {
    this.seedPersistentFleets();
  }

  private seedPersistentFleets() {
    // Seed Authoritative USSF Battle Formations
    this.fleets.set('ussf-vanguard-1', {
      fleetId: 'ussf-vanguard-1',
      faction: 'USSF',
      globalX: Math.fround(1024.0),
      globalY: Math.fround(1024.0),
      formationSize: 4,
      currentVelocity: createDeterministicVector(1.5, 0.5),
      targetSectorX: 8,
      targetSectorY: 8
    });

    // Seed Authoritative Accord Interceptor Squadrons
    this.fleets.set('accord-strike-6', {
      fleetId: 'accord-strike-6',
      faction: 'ACCORD',
      globalX: Math.fround(3072.0),
      globalY: Math.fround(3072.0),
      formationSize: 3,
      currentVelocity: createDeterministicVector(-1.0, -1.2),
      targetSectorX: 0,
      targetSectorY: 0
    });
  }

  /**
   * Evaluates persistent traffic movement independently from client clock.
   * Runs on a strict 2-minute cycle synchronized with server time tracking.
   */
  public advanceWorldTrafficSchedule() {
    this.fleets.forEach(fleet => {
      const dx = Math.fround(fleet.targetSectorX * ENGINE_CONFIG.CHUNK_SIZE - fleet.globalX);
      const dy = Math.fround(fleet.targetSectorY * ENGINE_CONFIG.CHUNK_SIZE - fleet.globalY);
      const distance = Math.fround(Math.sqrt(Math.fround(dx * dx + dy * dy)));

      if (distance < 200.0) {
        // Routinely cycle targets to generate continuous cruising loops
        fleet.targetSectorX = Math.floor(Math.random() * 16);
        fleet.targetSectorY = Math.floor(Math.random() * 16);
      } else {
        // Step vector coords cleanly matching server architecture limits
        const headingX = Math.fround(dx / distance);
        const headingY = Math.fround(dy / distance);
        fleet.currentVelocity = createDeterministicVector(headingX * 4.5, headingY * 4.5);

        fleet.globalX = Math.fround(fleet.globalX + fleet.currentVelocity.x);
        fleet.globalY = Math.fround(fleet.globalY + fleet.currentVelocity.y);
      }
    });
  }

  public getFleetsInProximity(x: number, y: number, radius: number): FactionFleet[] {
    const nearby: FactionFleet[] = [];
    this.fleets.forEach(fleet => {
      const dx = Math.fround(fleet.globalX - x);
      const dy = Math.fround(fleet.globalY - y);
      const distSq = Math.fround(dx * dx + dy * dy);
      if (distSq <= Math.fround(radius * radius)) {
        nearby.push({ ...fleet });
      }
    });
    return nearby;
  }
}

worldOpsRouter.post('/settle-encounter', async (req: Request, res: Response): Promise<void> => {
  try {

    const serverToken = "TRANS_SECRET_HANDSHAKE_KEY_TOKEN_73921";

    // 1. Cryptographic validation check gate
    const isSane = TelemetrySecurityGate.verifyPayloadSanity(payload, signature, serverToken);
    if (!isSane) {
      res.status(403).json({ error: "HTTP 403: Security validation handshake failed." });
      return;
    }

    // 2. Process authoritative flight path verification
    const simCheck = verifyFlightProximityAndSimulate({
      matchId: payload.matchId,
      pilotId: payload.pilotId,
      initialState: payload.initialState,
      inputFrames: payload.inputFrames,
      patrolTargetLocation: payload.patrolTargetLocation
    });

    if (!simCheck.isValid) {
      res.status(400).json({ error: simCheck.errorCode });
      return;
    }

    // 3. Update account details and reward metrics safely
    const profile = userProfileCache.get(payload.pilotId)!;
    profile.kills += 1;
    profile.credits += 45; // Update account balance cleanly

    res.status(200).json({
      success: true,
      newBalance: profile.credits,
      totalKillCount: profile.kills,
      msg: "Encounter resolved. Faction metrics synchronized."
    });

  } catch (error) {
    res.status(500).json({ error: "HTTP 500: Encounter resolver pipeline crashed." });
  }

3. Complete Orbital-Only 3D Traffic Renderer (artifacts/war-for-mars/src/game/orbital-native-flight-renderer.ts)

This high-performance module handles your complete Orbital-only migration. It removes all old 2D Canvas methods, using WebGL InstancedMesh clusters to draw faction spaceships as 3D traffic across the universe grid.
typescript
// artifacts/war-for-mars/src/game/orbital-native-flight-renderer.ts
import * as THREE from 'three';
import { FactionFleet } from '../../../api-server/src/game/universe-patrols';
import { ENGINE_CONFIG } from '../../../../lib/deterministic-core';

export class OrbitalNativeFlightRenderer {
  private scene: THREE.Scene;
  private ussfMeshInstances!: THREE.InstancedMesh;
  private accordMeshInstances!: THREE.InstancedMesh;
  private maxShipsPerFaction = 500;

  constructor(scene: THREE.Scene) {
    this.scene = scene;
    this.initializeFactionMeshes();
  }

  private initializeFactionMeshes() {
    // Structural 3D Configuration: USSF Fleet Ships
    const ussfGeom = new THREE.ConeGeometry(4, 14, 4);
    ussfGeom.rotateX(Math.PI / 2);
    const ussfMat = new THREE.MeshStandardMaterial({ color: '#4488ff', metalness: 0.7, roughness: 0.3 });
    this.ussfMeshInstances = new THREE.InstancedMesh(ussfGeom, ussfMat, this.maxShipsPerFaction);
    this.ussfMeshInstances.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
    this.scene.add(this.ussfMeshInstances);

    // Structural 3D Configuration: Accord Fleet Ships
    const accordGeom = new THREE.BoxGeometry(6, 3, 12);
    const accordMat = new THREE.MeshStandardMaterial({ color: '#ffaa00', metalness: 0.9, roughness: 0.1 });
    this.accordMeshInstances = new THREE.InstancedMesh(accordGeom, accordMat, this.maxShipsPerFaction);
    this.accordMeshInstances.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
    this.scene.add(this.accordMeshInstances);
  }

  /**
   * Renders the opposing faction's fleet as active, free-roaming traffic.
   * If the player pilot flies a USSF ship, Accord interceptor fleets populate the local grid map.
   */
  public updateSceneTraffic(playerFaction: 'USSF' | 'ACCORD', nearbyFleets: FactionFleet[], shipX: number, shipY: number) {
    const dummy = new THREE.Object3D();
    let ussfIdx = 0;
    let accordIdx = 0;

    nearbyFleets.forEach(fleet => {
      // Determine ship behavior based on selected faction alignment
      const isOpposingFleet = fleet.faction !== playerFaction;

      for (let i = 0; i < fleet.formationSize; i++) {
        // Apply formation sorting offsets cleanly using integer bounds
        const formationOffsetX = (i * 25) * Math.fround(Math.sin(i));
        const formationOffsetY = (i * 25) * Math.fround(Math.cos(i));

        const worldX = Math.fround(fleet.globalX + formationOffsetX);
        const worldY = Math.fround(fleet.globalY + formationOffsetY);

        dummy.position.set(worldX, isOpposingFleet ? 0 : 5, worldY);
        
        // Calculate heading vector paths
        const headingAngle = Math.atan2(fleet.currentVelocity.x, fleet.currentVelocity.y);
        dummy.rotation.set(0, headingAngle, 0);
        dummy.updateMatrix();

        if (fleet.faction === 'USSF' && ussfIdx < this.maxShipsPerFaction) {
          this.ussfMeshInstances.setMatrixAt(ussfIdx++, dummy.matrix);
        } else if (fleet.faction === 'ACCORD' && accordIdx < this.maxShipsPerFaction) {
          this.accordMeshInstances.setMatrixAt(accordIdx++, dummy.matrix);
        }
      }
    });

    // Clear unused transformation instances out of viewport bounds
    while (ussfIdx < this.maxShipsPerFaction) {
      dummy.position.set(0, -99999, 0);
      dummy.updateMatrix();
      this.ussfMeshInstances.setMatrixAt(ussfIdx++, dummy.matrix);
    }
    while (accordIdx < this.maxShipsPerFaction) {
      dummy.position.set(0, -99999, 0);
      dummy.updateMatrix();
      this.accordMeshInstances.setMatrixAt(accordIdx++, dummy.matrix);
    }

    this.ussfMeshInstances.instanceMatrix.needsUpdate = true;
    this.accordMeshInstances.instanceMatrix.needsUpdate = true;
  }
}

4. Unified Flight Deployment Hub Interface (artifacts/war-for-mars/src/game/GalaxyScreen.tsx)

This React hub connects your core code subsystems. It queries faction data from the server, configures the 3D rendering loops, and initializes the fallback recovery logic if the network connection breaks.
tsx
// artifacts/war-for-mars/src/game/GalaxyScreen.tsx
import React, { useEffect, useRef, useState } from 'react';
import * as THREE from 'three';
import { OrbitalNativeFlightRenderer } from './orbital-native-flight-renderer';
import { FactionFleet } from '../../../api-server/src/game/universe-patrols';

export const GalaxyScreen: React.FC = () => {
  const mountRef = useRef<HTMLDivElement | null>(null);
  const [pilotFaction, setPilotFaction] = useState<'USSF' | 'ACCORD'>('USSF');
  const [scoreTracker, setScoreTracker] = useState({ kills: 14, credits: 90 });

  const rendererRef = useRef<OrbitalNativeFlightRenderer | null>(null);

  useEffect(() => {
    if (!mountRef.current) return;

    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(60, 800 / 500, 0.1, 10000);
    const webGLRenderer = new THREE.WebGLRenderer({ antialias: true });
    webGLRenderer.setSize(800, 500);
    mountRef.current.appendChild(webGLRenderer.domElement);

    // Initialize the complete Orbital rendering migration manager instance
    rendererRef.current = new OrbitalNativeFlightRenderer(scene);

    camera.position.set(2048, 500, 2300);
    camera.lookAt(2048, 0, 2048);

    // Mock incoming server locations stream updates
    const mockNearbyFleets: FactionFleet[] = [
      { fleetId: 'acc-1', faction: 'ACCORD', globalX: 2048, globalY: 2000, formationSize: 3, currentVelocity: { x: 1, y: 1 }, targetSectorX: 4, targetSectorY: 4 },
      { fleetId: 'ussf-1', faction: 'USSF', globalX: 1800, globalY: 2200, formationSize: 4, currentVelocity: { x: -1, y: 0 }, targetSectorX: 2, targetSectorY: 2 }
    ];

    let animId: number;
    const animate = () => {
      if (rendererRef.current) {
        rendererRef.current.updateSceneTraffic(pilotFaction, mockNearbyFleets, 2048, 2048);
      }
      webGLRenderer.render(scene, camera);
      animId = requestAnimationFrame(animate);
    };
    animate();

    return () => {
      cancelAnimationFrame(animId);
      if (mountRef.current && webGLRenderer.domElement) {
        mountRef.current.removeChild(webGLRenderer.domElement);
      }
      webGLRenderer.dispose();
    };
  }, [pilotFaction]);

  return (
    <div className="w-full h-full min-h-screen bg-slate-950 p-6 flex flex-col items-center text-slate-100 font-mono">
      <div className="w-full max-w-4xl bg-slate-900 border border-slate-800 p-4 rounded-xl mb-4 flex justify-between items-center text-xs">
        <div>
          PILOT FACTION ALIGNMENT: 
          <select 
            value={pilotFaction} 
            onChange={(e) => setPilotFaction(e.target.value as any)}
            className="ml-2 bg-slate-950 border border-slate-800 text-red-400 p-1 rounded font-bold"
          >
            <option value="USSF">UNITED STATES SPACE FORCE (USSF)</option>
            <option value="ACCORD">ACCORD COMBINE</option>
          </select>
        </div>
        <div className="flex gap-4">
          <div>CONFIRMED KILLS: <span className="text-red-500 font-bold">{scoreTracker.kills}</span></div>
          <div>ACCOUNT STATUS: <span className="text-amber-400 font-bold">{scoreTracker.credits} CREDITS</span></div>
        </div>
      </div>

      <div ref={mountRef} className="w-full max-w-4xl h-[500px] border border-slate-800 rounded-xl overflow-hidden bg-black" />
    </div>
  );
};
