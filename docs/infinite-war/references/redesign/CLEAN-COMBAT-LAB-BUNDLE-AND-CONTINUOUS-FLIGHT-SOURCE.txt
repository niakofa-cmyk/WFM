import react from '@vitejs/plugin-react';

export default defineConfig({

  build: {
    target: 'esnext',

            return 'vendor-three-orbital';

          if (id.includes('node_modules/react') || id.includes('node_modules/react-dom')) {
            return 'vendor-react-core';

          if (id.includes('src/game/orbital-native')) {
            return 'engine-rendering-layer';
          }
          if (id.includes('lib/deterministic-core') || id.includes('lib/security-gate')) {

🛰️ Part 2: Continuous Server-Authoritative Free-Flight Matrix (artifacts/api-server/src/game/live-region-manager.ts)

Create this new orchestration module on your API layer to handle real-time spatial sector management. It segments active human pilots into distinct coordinate chunks, broadcasting position updates only to players sharing adjacent orbital sectors.
typescript
// artifacts/api-server/src/game/live-region-manager.ts
import { ENGINE_CONFIG, createDeterministicVector, Vector2D } from '../../../../lib/deterministic-core';

export interface HumanPilotSession {
  pilotId: number;
  faction: 'USSF' | 'ACCORD';
  x: number;
  y: number;
  heading: number;
  lastSeenTick: number;
}

export class LiveRegionManager {
  // Sector Map Key format: "chunkX,chunkY" -> Array of active pilot sessions
  private regionalSectors: Map<string, HumanPilotSession[]> = new Map();
  private pilotPositions: Map<number, { chunkKey: string; session: HumanPilotSession }> = new Map();

  private getSectorKey(x: number, y: number): string {
    const cx = Math.floor(x / ENGINE_CONFIG.CHUNK_SIZE);
    const cy = Math.floor(y / ENGINE_CONFIG.CHUNK_SIZE);
    return `${cx},${cy}`;
  }

  /**
   * Authoritatively stream and update pilot coordinates.
   * If a pilot crosses into an adjacent chunk, their session is migrated seamlessly.
   */
  public updatePilotTelemetry(pilotId: number, faction: 'USSF' | 'ACCORD', x: number, y: number, heading: number, tick: number): void {
    const targetX = Math.fround(x);
    const targetY = Math.fround(y);
    const targetHeading = Math.fround(heading);
    const newKey = this.getSectorKey(targetX, targetY);

    // Evict old position keys if a sector migration boundary is breached
    if (this.pilotPositions.has(pilotId)) {
      const oldRecord = this.pilotPositions.get(pilotId)!;
      if (oldRecord.chunkKey !== newKey) {
        const structuralList = this.regionalSectors.get(oldRecord.chunkKey) || [];
        this.regionalSectors.set(
          oldRecord.chunkKey, 
          structuralList.filter(p => p.pilotId !== pilotId)
        );
      }
    }

    const session: HumanPilotSession = { pilotId, faction, x: targetX, y: targetY, heading: targetHeading, lastSeenTick: tick };
    
    // Add to new chunk registry
    const currentSectorList = this.regionalSectors.get(newKey) || [];
    const existingIndex = currentSectorList.findIndex(p => p.pilotId === pilotId);
    if (existingIndex !== -1) {
      currentSectorList[existingIndex] = session;
    } else {
      currentSectorList.push(session);
    }
    
    this.regionalSectors.set(newKey, currentSectorList);
    this.pilotPositions.set(pilotId, { chunkKey: newKey, session });
  }

  /**
   * Queries 3x3 adjacent location blocks around a pilot.
   * Restricts network data delivery exclusively to what the player can actively detect.
   */
  public getLocalLiveTraffic(pilotId: number): HumanPilotSession[] {
    const tracking = this.pilotPositions.get(pilotId);
    if (!tracking) return [];

    const [cx, cy] = tracking.chunkKey.split(',').map(Number);
    const gatheredTraffic: HumanPilotSession[] = [];

    for (let dx = -1; dx <= 1; dx++) {
      for (let dy = -1; dy <= 1; dy++) {
        const queryKey = `${cx + dx},${cy + dy}`;
        const sectorPilots = this.regionalSectors.get(queryKey) || [];
        
        for (const targetPilot of sectorPilots) {
          if (targetPilot.pilotId !== pilotId) {
            gatheredTraffic.push(targetPilot);
          }
        }
      }
    }

    return gatheredTraffic;
  }

  public purgeInactivePilots(currentTick: number) {
    this.pilotPositions.forEach((value, pilotId) => {
      // Evict connections inactive for more than 180 server frames (3 seconds)
      if (currentTick - value.session.lastSeenTick > 180) {
        const list = this.regionalSectors.get(value.chunkKey) || [];
        this.regionalSectors.set(value.chunkKey, list.filter(p => p.pilotId !== pilotId));
        this.pilotPositions.delete(pilotId);
        console.log(`[Server Cleanup] Purged inactive pilot session node: ${pilotId}`);
      }
    });
  }
}

⚡ Part 3: Live Real-Time Multi-Entity Streaming Route (routes/world-operations.ts)

Update your server router to process live telemetry frames continuously. It handles real-time coordination, feeds your region manager, and computes cross-runtime delta verifications on every incoming call.
typescript
// artifacts/api-server/src/routes/world-operations.ts
import { Router, Request, Response } from 'express';
import { TelemetrySecurityGate } from '../../../../lib/security-gate';
import { LiveRegionManager } from '../game/live-region-manager';

export const worldOpsRouter = Router();
const regionManager = new LiveRegionManager();

let sharedGlobalServerTick = 0;

/**
 * High-Frequency Continuous Telemetry Synchronization Route
 * Validates, records, and streams position data for nearby human pilots.
 */
worldOpsRouter.post('/stream-telemetry', async (req: Request, res: Response): Promise<void> => {
  try {
    const { payload, signature } = req.body;
    sharedGlobalServerTick++;

    // 1. Verify integrity of the payload array stream
    const isSane = TelemetrySecurityGate.verifyPayloadSanity(payload, signature, "TRANS_SECRET_HANDSHAKE_KEY_TOKEN_73921");
    if (!isSane) {
      res.status(403).json({ error: "HTTP 403: Security handshakes reject malformed data arrays." });
      return;
    }

    // 2. Feed localized telemetry vectors into region engine
    regionManager.updatePilotTelemetry(
      payload.pilotId,
      payload.faction,
      payload.currentState.x,
      payload.currentState.y,
      payload.currentState.heading,
      sharedGlobalServerTick
    );

    // 3. Extract and return region-scoped traffic profiles
    const nearbyHumanTraffic = regionManager.getLocalLiveTraffic(payload.pilotId);

    // 4. Periodically run structural heap memory scrub sweeps
    if (sharedGlobalServerTick % 600 === 0) {
      regionManager.purgeInactivePilots(sharedGlobalServerTick);
    }

    res.status(200).json({
      success: true,
      serverTick: sharedGlobalServerTick,
      liveHumanTraffic: nearbyHumanTraffic.map(p => ({
        id: p.pilotId,
        f: p.faction,
        x: p.x,
        y: p.y,
        h: p.heading
      }))
    });

  } catch (error) {
    res.status(500).json({ error: "HTTP 500: Spatial orchestration state loop crashed." });
  }
});

// artifacts/war-for-mars/src/game/GalaxyScreen.tsx
import React, { useEffect, useRef, useState, Suspense } from 'react';

// Explicit dynamic split pattern cleanly separates the bundle allocation trees
const LazyThreeEngine = React.lazy(() => 
  import('./orbital-native-flight-renderer').then(module => ({
    default: function ThreeWrapper({ faction, traffic }: { faction: string; traffic: any[] }) {
      const mountRef = useRef<HTMLDivElement | null>(null);
      const rendererInstance = useRef<any>(null);
      const activeScene = useRef(new (require('three').Scene)());

      useEffect(() => {
        if (!mountRef.current) return;
        
        const THREE = require('three');
        const width = mountRef.current.clientWidth || 800;
        const height = mountRef.current.clientHeight || 500;
        
        const camera = new THREE.PerspectiveCamera(60, width / height, 0.1, 10000);
        const webGLRenderer = new THREE.WebGLRenderer({ antialias: true });
        webGLRenderer.setSize(width, height);
        mountRef.current.appendChild(webGLRenderer.domElement);

        const coreModule = require('./orbital-native-flight-renderer');
        rendererInstance.current = new coreModule.OrbitalNativeFlightRenderer(activeScene.current);

        camera.position.set(2048, 450, 2350);
        camera.lookAt(2048, 0, 2048);

        let activeId: number;
        const frameStep = () => {
          if (rendererInstance.current) {
            rendererInstance.current.updateSceneTraffic(faction, traffic, 2048, 2048);
          }
          webGLRenderer.render(activeScene.current, camera);
          activeId = requestAnimationFrame(frameStep);
        };
        activeId = requestAnimationFrame(frameStep);

        return () => {
          cancelAnimationFrame(activeId);
          if (mountRef.current && webGLRenderer.domElement) {
            mountRef.current.removeChild(webGLRenderer.domElement);
          }
          webGLRenderer.dispose();
        };
      }, [faction, traffic]);

      return <div ref={mountRef} className="w-full h-[500px] bg-black" />;
    }
  }))
);
