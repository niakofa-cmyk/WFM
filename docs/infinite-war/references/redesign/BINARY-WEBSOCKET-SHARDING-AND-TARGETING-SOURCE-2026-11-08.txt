💱 1. High-Frequency Binary WebSocket Sharding Server (artifacts/api-server/src/game/flight-socket-server.ts)

This high-performance engine replaces HTTP polling with an unthrottled, binary WebSocket network stream. It reads incoming raw input packets, updates the server-authoritative simulation space, and pushes anonymous, location-scoped snapshots down to connected pilots.
typescript
// artifacts/api-server/src/game/flight-socket-server.ts
import { WebSocketServer, WebSocket } from 'ws';
import { SharedFlightSessionHub } from './shared-flight-session';
import { TelemetrySecurityGate } from '../../../../lib/security-gate';

export class FlightSocketServer {
  private wss: WebSocketServer;
  private sessionHub: SharedFlightSessionHub;

  constructor(port: number) {
    this.wss = new WebSocketServer({ port });
    this.sessionHub = new SharedFlightSessionHub();
    this.initializeSocketEvents();
  }

  private initializeSocketEvents() {
    console.log(`[Matrix Socket] Streaming server active on port ${this.wss.options.port}`);

    this.wss.on('connection', (ws: WebSocket) => {
      let establishedPilotId: number | null = null;

      ws.on('message', (message: Buffer) => {
        try {
          // Parse packed binary message stream
          const view = new DataView(message.buffer, message.byteOffset, message.byteLength);
          const opCode = view.getUint8(0);

          // Client Input Stream OpCode Handling
          if (opCode === 1) {
            const pilotId = view.getUint32(1, true);
            const clientTick = view.getUint32(5, true);
            const bitfield = view.getUint8(9);

            if (!establishedPilotId) {
              establishedPilotId = pilotId;
              const factionId = view.getUint8(10);
              this.sessionHub.joinSession(pilotId, factionId === 1 ? 'USSF' : 'ACCORD', 2048, 2048);
            }

            const thrust = (bitfield & (1 << 0)) !== 0;
            const left = (bitfield & (1 << 1)) !== 0;
            const right = (bitfield & (1 << 2)) !== 0;

            // Step movement loops on server-authoritative simulation thread
            this.sessionHub.stepAuthoritativeMovement(pilotId, clientTick, { thrust, left, right });
            this.sessionHub.incrementServerClock();

            // Send local traffic snapshot back to client
            const snapshot = this.sessionHub.generateSpatialSnapshot(pilotId);
            const outBuffer = this.serializeSnapshotToBinary(snapshot);
            ws.send(Buffer.from(outBuffer));
          }
        } catch (err) {
          console.error('[Socket Exception] Frame parsing stalled:', err);
        }
      });

      ws.on('close', () => {
        if (establishedPilotId) {
          console.log(`[Socket Cleanup] Pilot ${establishedPilotId} connection matrix detached.`);
        }
      });
    });
  }

  private serializeSnapshotToBinary(snapshot: any): ArrayBuffer {
    const actorCount = snapshot.activePilots.length;
    const buffer = new ArrayBuffer(7 + actorCount * 14);
    const view = new DataView(buffer);

    view.setUint8(0, 2); // SERVER_WORLD_RECONCILE OpCode
    view.setUint32(1, snapshot.serverTick, true);
    view.setUint16(5, actorCount, true);

    let offset = 7;
    for (const p of snapshot.activePilots) {
      view.setUint32(offset, p.pilotId, true); // Masked proxy token identifier
      view.setUint8(offset + 4, p.faction === 'USSF' ? 1 : 2);
      view.setFloat32(offset + 5, p.position.x, true);
      view.setFloat32(offset + 9, p.position.y, true);
      view.setUint8(offset + 13, Math.floor((p.heading / 360) * 255));
      offset += 14;
    }
    return buffer;
  }
}

🎯 2. Continuous Sensor Lock-On Targeting Matrix (lib/sensor-targeting.ts)

To enable inter-ship tracking in free flight without causing garbage collection stutters, this system implements a Ray-Faceted Lock Acquisition Loop. It queries surrounding spatial actors, tests cross-vector cone geometries, and locks onto targets when they enter weapon range.
typescript
// lib/sensor-targeting.ts
import { Vector2D, MathUtils } from './deterministic-core';
import { LiveActorProxy } from '../game/orbital-native-flight-renderer';

export interface TargetingSystemState {
  targetAcquired: boolean;
  lockedProxyToken: number | null;
  lockProgress: number; // Scale from 0.0 to 1.0 (Fully Locked)
  relativeBearing: number;
}

export class SensorTargetingMatrix {
  private maxLockDistance = 1200.0; // Sensory envelope radius limits
  private lockConeRadians = Math.fround(30 * (Math.PI / 180)); // 30-degree tracking radar cone
  private lockAcquisitionRate = 0.05; // Lock speed step increment per frame tick

  /**
   * Tracks and resolves sensor locks against surrounding human ship vectors.
   */
  public evaluateLockStep(
    pilotX: number,
    pilotY: number,
    pilotHeadingDegrees: number,
    visibleActors: LiveActorProxy[],
    currentState: TargetingSystemState
  ): TargetingSystemState {
    const nextState = { ...currentState };
    const headingRad = Math.fround(pilotHeadingDegrees * (Math.PI / 180));

    // 1. Maintain lock focus if a target is already selected
    if (nextState.lockedProxyToken !== null) {
      const activeTarget = visibleActors.find(a => a.tokenProxy === nextState.lockedProxyToken);
      if (!activeTarget) {
        return { targetAcquired: false, lockedProxyToken: null, lockProgress: 0, relativeBearing: 0 };
      }

      const dx = Math.fround(activeTarget.x - pilotX);
      const dy = Math.fround(activeTarget.y - pilotY);
      const distance = Math.fround(Math.sqrt(Math.fround(dx * dx + dy * dy)));

      if (distance > this.maxLockDistance) {
        // Break lock if target leaves radar range
        return { targetAcquired: false, lockedProxyToken: null, lockProgress: 0, relativeBearing: 0 };
      }

      nextState.relativeBearing = Math.fround(Math.atan2(dx, dy) - headingRad);
      nextState.lockProgress = Math.min(1.0, Math.fround(nextState.lockProgress + this.lockAcquisitionRate));
      if (nextState.lockProgress >= 1.0) nextState.targetAcquired = true;

      return nextState;
    }

    // 2. Scan for valid new targets inside the front sensor radar cone
    let bestCandidate: LiveActorProxy | null = null;
    let narrowestAngle = this.lockConeRadians;

    for (const actor of visibleActors) {
      const dx = Math.fround(actor.x - pilotX);
      const dy = Math.fround(actor.y - pilotY);
      const distance = Math.fround(Math.sqrt(Math.fround(dx * dx + dy * dy)));

      if (distance > this.maxLockDistance) continue;

      const targetAngle = Math.fround(Math.atan2(dx, dy));
      let angleDiff = Math.abs(Math.fround(targetAngle - headingRad));
      
      // Normalize delta angles cleanly
      if (angleDiff > Math.PI) angleDiff = Math.fround(Math.fusing(Math.PI * 2) - angleDiff);

      if (angleDiff < narrowestAngle) {
        narrowestAngle = angleDiff;
        bestCandidate = actor;
      }
    }

    if (bestCandidate) {
      nextState.lockedProxyToken = bestCandidate.tokenProxy;
      nextState.lockProgress = this.lockAcquisitionRate;
      nextState.targetAcquired = false;
      nextState.relativeBearing = narrowestAngle;
    }

    return nextState;
  }
}

🎨 3. Real-Time Dynamic Target Retention HUD Overlay

We extend your OrbitalNativeFlightRenderer to project Sensor Tracking Ring brackets around targeted human ships in your 3D viewport canvas when a sensor lock is acquired.
typescript
// Extension integrated inside artifacts/war-for-mars/src/game/orbital-native-flight-renderer.ts

export class OrbitalNativeFlightRenderer {
  // ... Keep previous mesh instances and trail systems active
  private lockReticleMesh!: THREE.LineLoop;

  private initializeTargetingUI(scene: THREE.Scene) {
    const ringGeometry = new THREE.RingGeometry(12, 14, 6);
    const ringMaterial = new THREE.MeshBasicMaterial({ color: '#ff3333', side: THREE.DoubleSide, transparent: true, opacity: 0 });
    
    // Create line-loop target reticle brackets
    this.lockReticleMesh = new THREE.LineLoop(ringGeometry, ringMaterial);
    this.lockReticleMesh.rotateX(Math.PI / 2); // Lay flat on the orbital plane grid
    scene.add(this.lockReticleMesh);
  }

  /**
   * Tracks and projects visual targeting graphics around locked human ships.
   */
  public updateLockGraphics(targetToken: number | null, liveActors: LiveActorProxy[], isFullyLocked: boolean) {
    if (targetToken === null) {
      (this.lockReticleMesh.material as THREE.MeshBasicMaterial).opacity = 0.0; // Hide reticle
      return;
    }

    const trackedActor = liveActors.find(a => a.tokenProxy === targetToken);
    if (trackedActor) {
      this.lockReticleMesh.position.set(trackedActor.x, trackedActor.faction === 'USSF' ? 0.2 : 5.2, trackedActor.y);
      
      // Change color state dynamically based on lock completion progress
      const mat = this.lockReticleMesh.material as THREE.MeshBasicMaterial;
      mat.opacity = 0.85;
      mat.color.set(isFullyLocked ? '#ff2222' : '#ffff22'); // Red for locked, Yellow for acquiring
      
      // Spin the reticle mesh over elapsed times to give interactive system feedback
      this.lockReticleMesh.rotation.z += 0.05;
    }
  }
}
