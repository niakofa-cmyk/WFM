🛰️ 1. Authoritative Shared-Flight Session Hub (artifacts/api-server/src/game/shared-flight-session.ts)

This core server engine component manages spatial session clusters. It tracks every pilot's coordinates using strict single-precision geometry (Math.fround) and processes their movements authoritatively.

// artifacts/api-server/src/game/shared-flight-session.ts
import { ENGINE_CONFIG, createDeterministicVector, Vector2D, computeHeadingVector } from '../../../../lib/deterministic-core';

export interface AuthorizedPilotState {
  pilotId: number;
  faction: 'USSF' | 'ACCORD';
  position: Vector2D;
  velocity: Vector2D;
  heading: number;
  lastInputTick: number;
}

export interface SharedFlightSnapshot {
  serverTick: number;
  activePilots: AuthorizedPilotState[];
}

export class SharedFlightSessionHub {
  private sessions: Map<number, AuthorizedPilotState> = new Map();
  private currentServerTick = 0;

  /**
   * Registers or updates an authorized participant inside the spatial shard.
   */
  public joinSession(pilotId: number, faction: 'USSF' | 'ACCORD', x: number, y: number) {
    if (!this.sessions.has(pilotId)) {
      this.sessions.set(pilotId, {
        pilotId,
        faction,
        position: createDeterministicVector(x, y),
        velocity: createDeterministicVector(0, 0),
        heading: 0.0,
        lastInputTick: this.currentServerTick
      });
      console.log(`[Authoritative Shard] Pilot ${pilotId} secure flight state initiated.`);
    }
  }

  /**
   * Processes an incoming verified network input frame under strict single-precision rules.
   * Server ownership guarantees no client can fake velocity vector magnitudes.
   */
  public stepAuthoritativeMovement(pilotId: number, clientTick: number, inputs: { thrust: boolean; left: boolean; right: boolean }) {
    const pilot = this.sessions.get(pilotId);
    if (!pilot) return;

    // Enforce uniform delta heading transformations
    let nextHeading = pilot.heading;
    if (inputs.left)  nextHeading = Math.fround(nextHeading - 4.0);
    if (inputs.right) nextHeading = Math.fround(nextHeading + 4.0);
    nextHeading = Math.fround((nextHeading + 360.0) % 360.0);

    const lookVec = computeHeadingVector(nextHeading);
    let vx = pilot.velocity.x;
    let vy = pilot.velocity.y;

    if (inputs.thrust) {
      vx = Math.fround(vx + Math.fround(lookVec.x * 0.18));
      vy = Math.fround(vy + Math.fround(lookVec.y * 0.18));
    }
    
    // Apply server-side cosmic drag decay parameters
    vx = Math.fround(vx * 0.98);
    vy = Math.fround(vy * 0.98);

    pilot.position = createDeterministicVector(pilot.position.x + vx, pilot.position.y + vy);
    pilot.velocity = createDeterministicVector(vx, vy);
    pilot.heading = nextHeading;
    pilot.lastInputTick = clientTick;

    this.sessions.set(pilotId, pilot);
  }

  /**
   * Generates an anonymous, location-scoped snapshot payload for participants sharing the sector.
   */
  public generateSpatialSnapshot(requestingPilotId: number): SharedFlightSnapshot {
    const requester = this.sessions.get(requestingPilotId);
    if (!requester) return { serverTick: this.currentServerTick, activePilots: [] };

    const nearbyPilots: AuthorizedPilotState[] = [];
    
    this.sessions.forEach((pilot) => {
      if (pilot.pilotId === requestingPilotId) return;

      // Check distance using strict single-precision euclidean bounds
      const dx = Math.fround(pilot.position.x - requester.position.x);
      const dy = Math.fround(pilot.position.y - requester.position.y);
      const distanceSq = Math.fround(dx * dx + dy * dy);

      // Visibility envelope limited strictly to immediate 3x3 location-scoped chunks
      if (distanceSq <= Math.fround(ENGINE_CONFIG.CHUNK_SIZE * ENGINE_CONFIG.CHUNK_SIZE * 4.0)) {
        nearbyPilots.push({
          ...pilot,
          // Anonymize personal telemetry data matrices to fulfill privacy directives
          pilotId: pilot.pilotId ^ 0xABCDEF // Masked runtime token proxy
        });
      }
    });

    return {
      serverTick: this.currentServerTick,
      activePilots: nearbyPilots
    };
  }

  public incrementServerClock() {
    this.currentServerTick++;
  }
}

💱 2. High-Frequency Binary Snapshot Channel (artifacts/api-server/src/routes/flight-channel.ts)

This endpoint acts as your low-overhead data route. It ingests client input packets, maps them to the authoritative physics session, and outputs location-scoped delta buffers.

// artifacts/api-server/src/routes/flight-channel.ts
import { Router, Request, Response } from 'express';
import { SharedFlightSessionHub } from '../game/shared-flight-session';
import { TelemetrySecurityGate } from '../../../../lib/security-gate';

export const flightChannelRouter = Router();
const sessionHub = new SharedFlightSessionHub();

/**
 * High-frequency snapshot synchronization channel.
 * Authenticates the participant, processes input steps, and returns local traffic positions.
 */
flightChannelRouter.post('/sync-channel', async (req: Request, res: Response): Promise<void> => {
  try {
    const { payload, signature, inputFrame } = req.body;
    const transientSecret = "TRANS_SECRET_HANDSHAKE_KEY_TOKEN_73921";

    // 1. Verify cryptographic telemetry headers to prevent spoofed data
    const isSane = TelemetrySecurityGate.verifyPayloadSanity(payload, signature, transientSecret);
    if (!isSane) {
      res.status(403).json({ error: "HTTP 403: Cryptographic verification mismatch." });
      return;
    }

    const realPilotId = payload.pilotId;
    sessionHub.joinSession(realPilotId, payload.faction, payload.currentState.x, payload.currentState.y);

    // 2. Commit the input frame directly to the server-owned simulation state
    if (inputFrame) {
      sessionHub.stepAuthoritativeMovement(realPilotId, inputFrame.tick, {
        thrust: inputFrame.thrust,
        left: inputFrame.steerLeft,
        right: inputFrame.steerRight
      });
    }

    sessionHub.incrementServerClock();

    // 3. Extract proximity-scoped live pilot positions anonymously
    const snapshot = sessionHub.generateSpatialSnapshot(realPilotId);

    res.status(200).json({
      success: true,
      tick: snapshot.serverTick,
      liveActors: snapshot.activePilots.map(p => ({
        tokenProxy: p.pilotId,
        faction: p.faction,
        x: p.position.x,
        y: p.position.y,
        h: p.heading
      }))
    });

  } catch (error) {
    console.error("[Channel Error] Sync failed:", error);
    res.status(500).json({ error: "HTTP 500: Shared-flight transmission pipeline crash." });
  }
});

  h: number;

  // ... Existing structural mesh references (ussfMeshInstances, accordMeshInstances, thrusterTrails) remain fully active
  
  /**
   * Evaluates live multi-pilot snapshots and appends authorized human ships
   * into the active free-flight WebGL draw buffer.
   */
  public renderAuthoritativeHumanActors(playerFaction: 'USSF' | 'ACCORD', liveActors: LiveActorProxy[]) {
    const dummy3D = new THREE.Object3D();
    
    // Use high instance allocation offset pools to separate live humans from persistent AI traffic
    let humanUssfIdx = 100; 
    let humanAccordIdx = 100;

    liveActors.forEach((actor) => {
      dummy3D.position.set(actor.x, actor.faction === 'USSF' ? 0 : 5, actor.y);
      
      const headingRad = actor.h * (Math.PI / 180);
      dummy3D.rotation.set(0, headingRad, 0);
      dummy3D.updateMatrix();

      // Trigger thruster plume particles dynamically behind the moving human player ship proxy
      this.thrusterTrails.injectThrusterPlume(actor.x, actor.y, headingRad, actor.faction);

      if (actor.faction === 'USSF' && humanUssfIdx < this.maxShipsPerFaction) {
        this.ussfMeshInstances.setMatrixAt(humanUssfIdx++, dummy3D.matrix);
      } else if (actor.faction === 'ACCORD' && humanAccordIdx < this.maxShipsPerFaction) {
        this.accordMeshInstances.setMatrixAt(humanAccordIdx++, dummy3D.matrix);
      }
    });

    // Flag update buffers for the rendering loop pass
    this.ussfMeshInstances.instanceMatrix.needsUpdate = true;
    this.accordMeshInstances.instanceMatrix.needsUpdate = true;
  }
