🛠️ Part 1: Automated Railway Staging Rollout Script (scripts/deploy-staging.sh)

This shell script handles the deployment process. It runs tests, checks bundle size constraints, applies database changes, and triggers your Railway staging rollout container.
bash
#!/usr/bin/env bash
# scripts/deploy-staging.sh

set -e # Exit immediately if any individual script line reports an error

COLOR_CYAN='\033[0;36m'
COLOR_GREEN='\033[0;32m'
COLOR_RED='\033[0;31m'
COLOR_RESET='\033[0m'

echo -e "${COLOR_CYAN}[Deployment Shard] Starting workspace audit and staging rollout compilation...${COLOR_RESET}"

# 1. Run strict workspace-quality check suite
echo -e "${COLOR_CYAN}[Step 1/4] Running compilation testing suites...${COLOR_RESET}"
pnpm run test --filter combat-lab

# 2. Verify bundle size budget constraints (Enforce initial bundle < 140 KiB)
echo -e "${COLOR_CYAN}[Step 2/4] Verifying production bundle budget metrics...${COLOR_RESET}"
pnpm run build --filter combat-lab

BUNDLE_CHECK=$(find artifacts/war-for-mars/dist -name "index-*.js" -exec gzip -c {} \; | wc -c)
MAX_BUDGET=143360 # 140 KiB in bytes

if [ "$BUNDLE_CHECK" -gt "$MAX_BUDGET" ]; then
    echo -e "${COLOR_RED}🚨 CRITICAL AUDIT FAILURE: Initial JS size (${BUNDLE_CHECK} bytes) exceeds the 140 KiB threshold! Blocking rollout.${COLOR_RESET}"
    exit 1
fi
echo -e "${COLOR_GREEN}✓ Initial bundle budget approved. Size constraints pass.${COLOR_RESET}"

# 3. Apply safe additive database migration schemas
echo -e "${COLOR_CYAN}[Step 3/4] Triggering database schema updates...${COLOR_RESET}"
if [ -f "./scripts/migrations/02_add_faction_and_stats.sql" ]; then
    # Injecting environment variables directly inside the temporary migration runtime container
    PGPASSWORD=$STAGING_DB_PASSWORD psql -h $STAGING_DB_HOST -U $STAGING_DB_USER -d $STAGING_DB_NAME -f ./scripts/migrations/02_add_faction_and_stats.sql
    echo -e "${COLOR_GREEN}✓ Database table migrations applied successfully.${COLOR_RESET}"
else
    echo -e "${COLOR_RED}🚨 ERROR: SQL migration asset missing from worktree.${COLOR_RESET}"
    exit 1
fi

# 4. Push build triggers directly into Railway container stacks
echo -e "${COLOR_CYAN}[Step 4/4] Triggering Railway infrastructure rollout...${COLOR_RESET}"
railway up --environment staging

echo -e "${COLOR_GREEN}🚀 SUCCESS: Staging infrastructure rollout successfully deployed without service disruption.${COLOR_RESET}"

🗺️ Part 2: Faction Territory Influence Grid Engine (lib/territory-matrix.ts)

This structural tracking module divides Martian airspace into a \(16 \times 16\) grid matrix. It automatically calculates territory control percentages on the fly by evaluating server-verified battle tallies, mapping influence zones cleanly without leaking memory.
typescript
// lib/territory-matrix.ts
import { ENGINE_CONFIG } from './deterministic-core';

export interface TerritorySector {
  sectorX: number;
  sectorY: number;
  dominantFaction: 'USSF' | 'ACCORD' | 'NEUTRAL';
  ussfInfluence: number;  // Normalized scale from 0.0 to 1.0
  accordInfluence: number;
}

export class FactionTerritoryMatrixEngine {
  private totalSectorsCount = 16;
  private sectors: Map<string, TerritorySector> = new Map();

  constructor() {
    this.initializeNeutralMap();
  }

  private initializeNeutralMap() {
    for (let x = 0; x < this.totalSectorsCount; x++) {
      for (let y = 0; y < this.totalSectorsCount; y++) {
        const key = `${x},${y}`;
        this.sectors.set(key, {
          sectorX: x,
          sectorY: y,
          dominantFaction: 'NEUTRAL',
          ussfInfluence: 0.33,
          accordInfluence: 0.33
        });
      }
    }
  }

  /**
   * Recalculates sector metrics following a server-verified kill.
   * Enforces strict single-precision floating-point bounds during calculations.
   */
  public registerCombatOutcome(globalX: number, globalY: number, victorFaction: 'USSF' | 'ACCORD') {
    const cx = Math.floor(globalX / ENGINE_CONFIG.CHUNK_SIZE);
    const cy = Math.floor(globalY / ENGINE_CONFIG.CHUNK_SIZE);
    const key = `${cx % this.totalSectorsCount},${cy % this.totalSectorsCount}`;

    const sector = this.sectors.get(key);
    if (!sector) return;

    // Apply incremental faction shift weights safely
    if (victorFaction === 'USSF') {
      sector.ussfInfluence = Math.fround(Math.min(1.0, sector.ussfInfluence + 0.08));
      sector.accordInfluence = Math.fround(Math.max(0.0, sector.accordInfluence - 0.04));
    } else {
      sector.accordInfluence = Math.fround(Math.min(1.0, sector.accordInfluence + 0.08));
      sector.ussfInfluence = Math.fround(Math.max(0.0, sector.ussfInfluence - 0.04));
    }

    // Evaluate absolute grid sector dominance thresholds
    const totalWeight = Math.fround(sector.ussfInfluence + sector.accordInfluence);
    if (totalWeight > 0) {
      const bias = Math.fround(sector.ussfInfluence / totalWeight);
      if (bias > 0.60) sector.dominantFaction = 'USSF';
      else if (bias < 0.40) sector.dominantFaction = 'ACCORD';
      else sector.dominantFaction = 'NEUTRAL';
    }

    this.sectors.set(key, sector);
  }

  public getSectorState(x: number, y: number): TerritorySector | undefined {
    return this.sectors.get(`${x},${y}`);
  }
}

🎨 Part 3: Reactive Galaxy Screen Tactical Dashboard Extension

We update your GalaxyScreen interface module to map hex grid overlays over the 3D viewport canvas. Cells change color automatically to reflect shifting battle lines in real-time.
tsx
// Extension integrated inside artifacts/war-for-mars/src/game/GalaxyScreen.tsx

import React, { useState } from 'react';
import { FactionTerritoryMatrixEngine } from '../../../../lib/territory-matrix';

export const GalaxyTerritoryOverlay: React.FC = () => {
  const [matrixEngine] = useState(new FactionTerritoryMatrixEngine());
  
  // Create an array list of the 16x16 grid elements for UI tracking loops
  const gridCells = Array.from({ length: 16 * 16 }, (_, i) => {
    const x = i % 16;
    const y = Math.floor(i / 16);
    return matrixEngine.getSectorState(x, y)!;
  });

  return (
    <div className="w-full max-w-4xl bg-slate-900 border border-slate-800 rounded-xl p-4 mt-6 font-mono text-xs text-slate-200">
      <div className="text-xs uppercase font-bold tracking-widest text-slate-400 mb-3 flex justify-between items-center">
        <span>🛰️ ORBITAL SECTOR INFLUENCE MAP MATRIX</span>
        <div className="flex gap-4 text-[10px]">
          <span className="flex items-center gap-1"><div className="w-2 h-2 bg-blue-600 rounded" /> USSF DOMINANT</span>
          <span className="flex items-center gap-1"><div className="w-2 h-2 bg-amber-500 rounded" /> ACCORD CONTROL</span>
          <span className="flex items-center gap-1"><div className="w-2 h-2 bg-slate-700 rounded" /> CONTESTED</span>
        </div>
      </div>

      {/* Hex Grid Matrix Display Container */}
      <div className="grid grid-cols-16 gap-1 bg-slate-950 p-3 rounded-lg border border-slate-800 overflow-x-auto">
        {gridCells.map((cell, idx) => (
          <div
            key={idx}
            className={`w-full aspect-square rounded transition-all duration-300 relative group cursor-crosshair border ${
              cell.dominantFaction === 'USSF' 
                ? 'bg-blue-950/60 border-blue-600 text-blue-400' 
                : cell.dominantFaction === 'ACCORD' 
                ? 'bg-amber-950/60 border-amber-500 text-amber-400' 
                : 'bg-slate-900/40 border-slate-800 text-slate-600'
            }`}
          >
            {/* Context Tooltip Element Hover Hook */}
            <div className="opacity-0 group-hover:opacity-100 transition-opacity absolute bottom-full left-1/2 -translate-x-1/2 bg-slate-950 border border-slate-800 p-2 rounded text-[10px] w-32 pointer-events-none z-50 shadow-xl text-slate-200">
              <div className="font-bold text-slate-400">Sector Core: [{cell.sectorX}, {cell.sectorY}]</div>
              <div className="mt-1">USSF IP: {(cell.ussfInfluence * 100).toFixed(0)}%</div>
              <div>ACRD IP: {(cell.accordInfluence * 100).toFixed(0)}%</div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
};
