🛡️ Part 2: 3D Shield Vector Flash System (artifacts/war-for-mars/src/game/orbital-native-flight-renderer.ts)

We extend your OrbitalNativeFlightRenderer to layer dynamic, glowing geometric vector bubbles around ships. By modifying uniform parameters instead of rebuilding geometry structures frame-by-frame, the WebGL pipeline runs at full speed without stalling the CPU thread.
typescript
// Additions integrated inside artifacts/war-for-mars/src/game/orbital-native-flight-renderer.ts

export class ShieldVectorFlashSystem {
  private shieldGeometry: THREE.SphereGeometry;
  private shieldMaterial: THREE.MeshBasicMaterial;
  private activeShieldMeshes: Map<number, THREE.Mesh> = new Map();
  private scene: THREE.Scene;

  constructor(scene: THREE.Scene) {
    this.scene = scene;
    // Low-polygon sphere bounding matrix encapsulating the fighter hulls
    this.shieldGeometry = new THREE.SphereGeometry(9, 16, 16);
    
    this.shieldMaterial = new THREE.MeshBasicMaterial({
      color: '#00f0ff',
      transparent: true,
      opacity: 0.0, // Hidden until damage absorption is triggered
      blending: THREE.AdditiveBlending,
      wireframe: true,
      depthWrite: false
    });
  }

  /**
   * Triggers a momentary high-intensity plasma glow around a targeted pilot ID.
   */
  public triggerShieldImpact(pilotId: number, faction: 'USSF' | 'ACCORD', positionX: number, positionZ: number) {
    if (!this.activeShieldMeshes.has(pilotId)) {
      const cloneMat = this.shieldMaterial.clone();
      // Theme shield hues matching native faction aesthetics
      cloneMat.color.set(faction === 'USSF' ? '#00f0ff' : '#ffaa00');
      
      const mesh = new THREE.Mesh(this.shieldGeometry, cloneMat);
      this.scene.add(mesh);
      this.activeShieldMeshes.set(pilotId, mesh);
    }

    const targetMesh = this.activeShieldMeshes.get(pilotId)!;
    targetMesh.position.set(positionX, faction === 'USSF' ? 0 : 5, positionZ);
    (targetMesh.material as THREE.MeshBasicMaterial).opacity = 0.85; // Flash energy immediately
  }

  /**
   * Gradually decays shield line opacity over elapsed frame times.
   */
  public stepShieldLifecycles() {
    this.activeShieldMeshes.forEach((mesh, pilotId) => {
      const mat = mesh.material as THREE.MeshBasicMaterial;
      if (mat.opacity > 0.0) {
        mat.opacity = Math.fround(mat.opacity - 0.04); // Smooth exponential energy decay
      } else {
        // Safe visibility culling
        mesh.position.set(0, -99999, 0);
      }
    });
  }

  public dispose() {
    this.activeShieldMeshes.forEach(mesh => {
      (mesh.material as THREE.MeshBasicMaterial).dispose();
      this.scene.remove(mesh);
    });
    this.shieldGeometry.dispose();
    this.shieldMaterial.dispose();
  }
}
