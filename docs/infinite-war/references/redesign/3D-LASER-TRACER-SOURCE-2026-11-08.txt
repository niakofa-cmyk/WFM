
🛰️ 3D Instantaneous Ray-Casted Laser Tracer Subsystem

To add a highly responsive weapon fire visualization layer to War for Mars: Infinite War without causing garbage collection drops or rendering pipeline stalls, we implement a pre-allocated 3D Weapon Fire Tracing Subsystem.
Instead of adding individual dynamic meshes frame-by-frame, this architecture utilizes a single unified THREE.LineSegments mesh running on top of a cyclic array buffer. When a ship fires a weapon, a line segment is instantly drawn between the firing entity's origin and its calculated impact point, decaying exponentially over a few frames to look like a glowing plasma trail.
[Server Weapon Fire Packet] ──> Decodes Core Origin / Impact Vector Coordinates
                                            │
                                            ▼ (Direct Heap Bypass Pass)
[Cyclic Array Ring Buffer]  ──> Updates Float32Array Position Attributes Inline
                                            │
                                            ▼
[WebGL Pipeline GPU Draw]   ──> Renders Highly Intense Additive Line Segments

1. 3D Laser Tracer Graphics Core (artifacts/war-for-mars/src/game/orbital-native-weapon-tracer.ts)

Create this new orchestration module within your 3D engine workspace. It handles position buffers directly on the GPU, drawing clear laser paths without impacting CPU performance.
typescript
// artifacts/war-for-mars/src/game/orbital-native-weapon-tracer.ts
import * as THREE from 'three';

export interface LaserTracerEvent {
  originX: number;
  originZ: number;
  targetX: number;
  targetZ: number;
  faction: 'USSF' | 'ACCORD';
}

export class OrbitalNativeWeaponTracer {
  private maxTracers = 150; // Simultaneous laser paths allowed in viewport bounds
  private pointsPerTracer = 2; // Start and End vertices
  private coordsPerPoint = 3;  // X, Y, Z

  private lineGeometry: THREE.BufferGeometry;
  private lineMaterial: THREE.LineBasicMaterial;
  private lineSegments: THREE.LineSegments;

  private positionsArray: Float32Array;
  private lifeArray: Float32Array; // Tracks visual fading lifecycles per laser line
  private ringIndex = 0;

  constructor(scene: THREE.Scene) {
    const totalFloats = this.maxTracers * this.pointsPerTracer * this.coordsPerPoint;
    this.positionsArray = new Float32Array(totalFloats);
    this.lifeArray = new Float32Array(this.maxTracers);

    // Initialize all lines offscreen to avoid runtime allocations
    for (let i = 0; i < totalFloats; i += 3) {
      this.positionsArray[i] = 0.0;
      this.positionsArray[i + 1] = -99999.0;
      this.positionsArray[i + 2] = 0.0;
    }

    this.lineGeometry = new THREE.BufferGeometry();
    this.lineGeometry.setAttribute('position', new THREE.BufferAttribute(this.positionsArray, 3));

    this.lineMaterial = new THREE.LineBasicMaterial({
      color: '#ff3344',
      transparent: true,
      opacity: 0.9,
      blending: THREE.AdditiveBlending,
      linewidth: 2, // Note: Linewidth > 1 is hardware/OS restricted in WebGL native, thickness handled via glow intensities
      depthWrite: false
    });

    this.lineSegments = new THREE.LineSegments(this.lineGeometry, this.lineMaterial);
    scene.add(this.lineSegments);
  }

  /**
   * Injects a raw instantaneous ray-cast path directly into the pre-allocated Float32 ring buffer.
   */
  public spawnWeaponTracer(event: LaserTracerEvent) {
    const laserId = this.ringIndex;
    const vertexOffset = laserId * this.pointsPerTracer * this.coordsPerPoint;

    // Faction-scoped altitude constraints
    const altitude = event.faction === 'USSF' ? 0.0 : 5.0;

    // Vertex 1: Weapon Muzzle Origin point
    this.positionsArray[vertexOffset]     = Math.fround(event.originX);
    this.positionsArray[vertexOffset + 1] = Math.fround(altitude);
    this.positionsArray[vertexOffset + 2] = Math.fround(event.originZ);

    // Vertex 2: Impact Ray-Cast target endpoint
    this.positionsArray[vertexOffset + 3] = Math.fround(event.targetX);
    this.positionsArray[vertexOffset + 4] = Math.fround(altitude);
    this.positionsArray[vertexOffset + 5] = Math.fround(event.targetZ);

    this.lifeArray[laserId] = 1.0; // Reset alpha lifecycle opacity to max intensity

    // Increment cyclic ring index counter bounds
    this.ringIndex = (this.ringIndex + 1) % this.maxTracers;
    this.lineGeometry.attributes.position.needsUpdate = true;
  }

  /**
   * Steps opacity decay profiles sequentially inside the requestAnimationFrame game loop.
   */
  public updateTracersLifecycle() {
    let updateNeeded = false;

    for (let i = 0; i < this.maxTracers; i++) {
      if (this.lifeArray[i] > 0.0) {
        this.lifeArray[i] = Math.fround(this.lifeArray[i] - 0.08); // Decay visibility scale over steps
        updateNeeded = true;

        if (this.lifeArray[i] <= 0.0) {
          this.lifeArray[i] = 0.0;
          const vertexOffset = i * this.pointsPerTracer * this.coordsPerPoint;
          
          // Move segment out of view limits once expired
          this.positionsArray[vertexOffset + 1] = -99999.0;
          this.positionsArray[vertexOffset + 4] = -99999.0;
        }
      }
    }

    if (updateNeeded) {
      this.lineGeometry.attributes.position.needsUpdate = true;
    }
  }

  public dispose() {
    this.lineGeometry.dispose();
    this.lineMaterial.dispose();
  }
}

import { OrbitalNativeWeaponTracer, LaserTracerEvent } from './orbital-native-weapon-tracer';

export class OrbitalNativeFlightRenderer {
  // ... Keep previous mesh instances, particle trails, and shield systems active
  private weaponTracerSystem!: OrbitalNativeWeaponTracer;

  // Append initialization step cleanly inside your core constructor loop context
  private initializeWeaponVisualizer(scene: THREE.Scene) {
    this.weaponTracerSystem = new OrbitalNativeWeaponTracer(scene);
  }

  public updateSceneTraffic(playerFaction: 'USSF' | 'ACCORD', nearbyFleets: any[], shipX: number, shipY: number) {
    // ... prior tracking calculation tasks proceed normally

    // Advance tracer life vectors before frame buffer submission passes
    if (this.weaponTracerSystem) {
      this.weaponTracerSystem.updateTracersLifecycle();
    }
    
    this.ussfMeshInstances.instanceMatrix.needsUpdate = true;
    this.accordMeshInstances.instanceMatrix.needsUpdate = true;
  }

  /**
   * Safe public endpoint method called whenever an inbound network weapon fire 
   * data packet clears the binary decompression filters.
   */
  public registerLaserFire(originX: number, originZ: number, targetX: number, targetZ: number, faction: 'USSF' | 'ACCORD') {
    this.weaponTracerSystem.spawnWeaponTracer({
      originX,
      originZ,
      targetX,
      targetZ,
      faction
    });
  }
}

3. Binary Communication Routing Extension (artifacts/api-server/src/game/flight-socket-server.ts)

Update your network messaging system to support an explicit OpCode 3: PROJECTILE_BURST packet. This broadcasts firing event vectors out to all adjacent pilots sharing the sector grid box.
typescript
// Additions integrated inside artifacts/api-server/src/game/flight-socket-server.ts

/**
 * Extension method to broadcast a compressed laser fire event packet.
 * Payload Size: 18 Bytes total (1 OpCode, 4 Origin X, 4 Origin Y, 4 Target X, 4 Target Y, 1 Faction bit)
 */
public broadcastLaserFireEvent(
  senderSocket: WebSocket,
  ox: number,
  oy: number,
  tx: number,
  ty: number,
  faction: 'USSF' | 'ACCORD'
) {
  const buffer = new ArrayBuffer(18);
  const view = new DataView(buffer);

  view.setUint8(0, 3); // OpCode 3: PROJECTILE_BURST
  view.setFloat32(1, Math.fround(ox), true);
  view.setFloat32(5, Math.fround(oy), true);
  view.setFloat32(9, Math.fround(tx), true);
  view.setFloat32(13, Math.fround(ty), true);
  view.setUint8(17, faction === 'USSF' ? 1 : 2);

  // Broadcast layout packet stream out to all connection endpoints except the sender
  this.wss.clients.forEach((client) => {
    if (client !== senderSocket && client.readyState === WebSocket.OPEN) {
      client.send(Buffer.from(buffer));
    }
  });
}
