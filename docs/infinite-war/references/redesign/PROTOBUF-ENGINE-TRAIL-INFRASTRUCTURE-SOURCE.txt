export class FleetThrusterParticleSystem {
  private particleGeometry: THREE.BufferGeometry;
  private particleMaterial: THREE.PointsMaterial;
  private particlePoints: THREE.Points;
  
  private maxParticles = 2000;
  private positionsArray: Float32Array;
  private velocitiesArray: Float32Array;
  private currentParticleCount = 0;

  constructor(scene: THREE.Scene) {
    this.positionsArray = new Float32Array(this.maxParticles * 3);
    this.velocitiesArray = new Float32Array(this.maxParticles * 3);

    this.particleGeometry = new THREE.BufferGeometry();
    this.particleGeometry.setAttribute('position', new THREE.BufferAttribute(this.positionsArray, 3));

    // High-intensity additive blending for plasma exhaustion effects
    this.particleMaterial = new THREE.PointsMaterial({
      color: '#00ffff',
      size: 3.5,
      transparent: true,
      opacity: 0.8,
      blending: THREE.AdditiveBlending,
      depthWrite: false
    });

    this.particlePoints = new THREE.Points(this.particleGeometry, this.particleMaterial);
    scene.add(this.particlePoints);
  }

  /**
   * Spawns an atomic exhaust plume coordinate directly behind a target hull vector.
   */
  public injectThrusterPlume(originX: number, originZ: number, headingRad: number, faction: 'USSF' | 'ACCORD') {
    const idx = this.currentParticleCount * 3;
    if (this.currentParticleCount >= this.maxParticles) {
      this.currentParticleCount = 0; // Wrap buffer allocation ring cleanly
    }

    // Offset point to emerge explicitly out of the rear exhaust manifolds
    const cosH = Math.cos(headingRad);
    const sinH = Math.sin(headingRad);
    
    this.positionsArray[idx] = Math.fround(originX - sinH * 8.0);
    this.positionsArray[idx + 1] = Math.fround(faction === 'USSF' ? 0.0 : 5.0);
    this.positionsArray[idx + 2] = Math.fround(originZ - cosH * 8.0);

    // Eject trail vectors in the precise opposite direction of flight momentum
    this.velocitiesArray[idx] = Math.fround(-sinH * 2.5 + (Math.random() - 0.5));
    this.velocitiesArray[idx + 1] = Math.fround((Math.random() - 0.5) * 0.2);
    this.velocitiesArray[idx + 2] = Math.fround(-cosH * 2.5 + (Math.random() - 0.5));

    this.currentParticleCount++;
  }

  /**
   * Updates and updates particle lifecycles sequentially across frames.
   */
  public stepParticlePhysics() {
    const positions = this.particleGeometry.attributes.position.array as Float32Array;

    for (let i = 0; i < this.maxParticles; i++) {
      const idx = i * 3;
      // Skip inactive or dead coordinate points
      if (positions[idx + 1] < -50000) continue;

      // Apply constant expansion velocity steps
      positions[idx] = Math.fround(positions[idx] + this.velocitiesArray[idx]);
      positions[idx + 1] = Math.fround(positions[idx + 1] + this.velocitiesArray[idx + 1]);
      positions[idx + 2] = Math.fround(positions[idx + 2] + this.velocitiesArray[idx + 2]);

      // Apply visual fading by bleeding down positions beneath the map floor mesh
      this.velocitiesArray[idx] *= 0.96;
      this.velocitiesArray[idx + 2] *= 0.96;
      
      if (Math.abs(this.velocitiesArray[idx]) < 0.1) {
        positions[idx + 1] = -99999; // Expire element out of loop context
      }
    }

    this.particleGeometry.attributes.position.needsUpdate = true;
  }
}

export class OrbitalNativeFlightRenderer {
  private scene: THREE.Scene;
  private thrusterTrails: FleetThrusterParticleSystem;
  // ... prior variable states retain operational continuity

  constructor(scene: THREE.Scene) {
    this.scene = scene;
    this.initializeFactionMeshes();
    this.thrusterTrails = new FleetThrusterParticleSystem(scene);
  }

  public updateSceneTraffic(playerFaction: 'USSF' | 'ACCORD', nearbyFleets: any[], shipX: number, shipY: number) {
    const dummy = new THREE.Object3D();
    let ussfIdx = 0;
    let accordIdx = 0;

    nearbyFleets.forEach(fleet => {
      const isOpposingFleet = fleet.faction !== playerFaction;
      const headingAngle = Math.atan2(fleet.currentVelocity.x, fleet.currentVelocity.y);

      for (let i = 0; i < fleet.formationSize; i++) {
        const formationOffsetX = (i * 25) * Math.fround(Math.sin(i));
        const formationOffsetY = (i * 25) * Math.fround(Math.cos(i));

        const worldX = Math.fround(fleet.globalX + formationOffsetX);
        const worldY = Math.fround(fleet.globalY + formationOffsetY);

        dummy.position.set(worldX, isOpposingFleet ? 0 : 5, worldY);
        dummy.rotation.set(0, headingAngle, 0);
        dummy.updateMatrix();

        // Inject active trailing exhaust particles directly into buffer arrays
        this.thrusterTrails.injectThrusterPlume(worldX, worldY, headingAngle, fleet.faction);

        if (fleet.faction === 'USSF' && ussfIdx < this.maxShipsPerFaction) {
          this.ussfMeshInstances.setMatrixAt(ussfIdx++, dummy.matrix);
        } else if (fleet.faction === 'ACCORD' && accordIdx < this.maxShipsPerFaction) {
          this.accordMeshInstances.setMatrixAt(accordIdx++, dummy.matrix);
        }
      }
    });

    // Run underlying particle physics loops
    this.thrusterTrails.stepParticlePhysics();

    // Clean remaining indices...
    this.ussfMeshInstances.instanceMatrix.needsUpdate = true;
    this.accordMeshInstances.instanceMatrix.needsUpdate = true;
  }
}
