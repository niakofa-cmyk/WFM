💾 2. Local Replay Cache & Fault Recovery Layer (lib/local-memory-cache.ts)

To protect your matches against HTTP 400 claim rejections or unexpected network drops, this module caches your raw input histories locally. It dumps frames cleanly to the .agents/memory/ directory via Node or an automated browser file export stream.
typescript
// lib/local-memory-cache.ts
import { UserInput, PlayerPhysicsState } from './network-sync';

export interface LocalMatchSnapshot {
  matchId: string;
  timestamp: number;
  totalFrames: number;
  initialConfiguration: { x: number; y: number; heading: number };
  capturedInputs: UserInput[];
  finalStateTrace: PlayerPhysicsState;
}

export class LocalMemoryCacheEngine {
  private memoryDirectory = './.agents/memory';

  /**
   * Flushes operational frame history instantly if a server error happens.
   * Keeps raw telemetry safe for local differential debugging scripts.
   */
  public dumpEmergencySnapshot(snapshot: LocalMatchSnapshot): boolean {
    try {
      const fileName = `emergency_match_${snapshot.matchId}_${Date.now()}.json`;
      const dataPayload = JSON.stringify(snapshot, null, 2);

      // Node.js Execution Context Save Path
      if (typeof window === 'undefined') {
        const fs = require('fs');
        const path = require('path');
        const fullPath = path.join(this.memoryDirectory, fileName);
        
        if (!fs.existsSync(this.memoryDirectory)) {
          fs.mkdirSync(this.memoryDirectory, { recursive: true });
        }
        
        fs.writeFileSync(fullPath, dataPayload, 'utf8');
        console.warn(`[Local Recovery Node Cache] Emergency match snapshot flushed cleanly: ${fullPath}`);
        return true;
      } else {
        // Browser Fallback Mode: Trigger an automated download dump file attachment
        const blob = new Blob([dataPayload], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = fileName;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        URL.revokeObjectURL(url);
        console.warn('[Local Recovery Browser Cache] Network disconnect caught. Replay telemetry file triggered.');
        return true;
      }
    } catch (error) {
      console.error('[Recovery Error Failed] Could not securely log match state trace history:', error);
      return false;
    }
  }
}

export const ProductionUniverseBuild: React.FC = () => {
  const [connectionStatus, setConnectionStatus] = useState<'OFFLINE' | 'CONNECTED' | 'RECOVERY_TRIGGERED'>('OFFLINE');
  const [trackedTick, setTrackedTick] = useState<number>(0);
  
  const socketRef = useRef<WebSocket | null>(null);
  const cacheEngineRef = useRef(new LocalMemoryCacheEngine());
  
  // Track inputs internally to back up local memory sessions safely
  const localInputLogRef = useRef<any[]>([]);
  const currentTickRef = useRef<number>(0);

  useEffect(() => {
    // Connect websocket pipeline directly to your validation channel loop
    const ws = new WebSocket('ws://localhost:8080');
    socketRef.current = ws;

    ws.onopen = () => {
      setConnectionStatus('CONNECTED');
      console.log('[Network Node Linked] Connected to server validation matrix.');
    };

    ws.onclose = () => {
      if (currentTickRef.current > 10) {
        setConnectionStatus('RECOVERY_TRIGGERED');
        triggerEmergencyBackup();
      } else {
        setConnectionStatus('OFFLINE');
      }
    };

    return () => {
      ws.close();
    };
