1. Server-Authoritative Global Campaign Directive (artifacts/api-server/src/game/global-campaign-director.ts)

This core tracking engine manages the universal story clock. It uses Redis hashes to maintain a single, cross-instance completion variable, automatically shifting available sectors and locking operations globally based on real-time multiplayer success.
typescript
// artifacts/api-server/src/game/global-campaign-director.ts
import { createClient } from 'redis';
import { Pool } from 'pg';

export interface UniverseCampaignState {
  currentGlobalProgressPercent: number;
  totalOperationsCleared: number;
  activePhase: 'INITIAL_CONTACT' | 'BEACON_RECON' | 'WARDEN_INTERCEPT' | 'CAGE_CLIMAX' | 'SYSTEM_VICTORY';
  isUniverseConcluded: boolean;
}

export class GlobalCampaignDirector {
  private redisClient;
  private dbPool: Pool;
  private targetMaxOperations = 50000; // Total collective operations required for 100% completion

  constructor(redisUrl: string, dbPool: Pool) {
    this.redisClient = createClient({ url: redisUrl });
    this.dbPool = dbPool;
  }

  public async initCampaignDirector() {
    await this.redisClient.connect();
    console.log("[Global Campaign Engine] Authoritative Storyline Clock linked successfully via Redis.");
  }

  /**
   * Evaluates collective player successes and updates the unified global progress scale.
   * Runs inside an atomic transaction to guarantee state parity across horizontal servers.
   */
  public async registerOperationClear(operationId: string): Promise<UniverseCampaignState> {
    const progressKey = 'wfm:campaign:global_state';

    // 1. Atomically increment the universal clearing metrics counter
    const currentCleared = await this.redisClient.hIncrBy(progressKey, 'totalOperationsCleared', 1);
    
    // 2. Compute floating-point progress scale with single-precision float accuracy
    let calculatedPercent = Math.fround((currentCleared / this.targetMaxOperations) * 100.0);
    if (calculatedPercent > 100.0) calculatedPercent = 100.0;

    // 3. Evaluate active structural phase transitions across the shared timeline
    let activePhase: UniverseCampaignState['activePhase'] = 'INITIAL_CONTACT';
    if (calculatedPercent >= 100.0) activePhase = 'SYSTEM_VICTORY';
    else if (calculatedPercent >= 75.0) activePhase = 'CAGE_CLIMAX';
    else if (calculatedPercent >= 50.0) activePhase = 'WARDEN_INTERCEPT';
    else if (calculatedPercent >= 25.0) activePhase = 'BEACON_RECON';

    const isUniverseConcluded = activePhase === 'SYSTEM_VICTORY';

    // 4. Update the distributed hash layer parameters
    await this.redisClient.hSet(progressKey, {
      currentGlobalProgressPercent: calculatedPercent.toString(),
      activePhase,
      isUniverseConcluded: isUniverseConcluded.toString()
    });

    // 5. Commit permanent historical checkpoints down to PostgreSQL every 50 operations cleared
    if (currentCleared % 50 === 0) {
      await this.dbPool.query(
        `UPDATE campaign_global_metadata SET progress_percent = $1, current_phase = $2, is_concluded = $3, updated_at = CURRENT_TIMESTAMP;`,
        [calculatedPercent, activePhase, isUniverseConcluded]
      );
      console.log(`[Database Sync] Persistent Campaign State checkpointed: ${calculatedPercent.toFixed(2)}% Complete.`);
    }

    return {
      currentGlobalProgressPercent: calculatedPercent,
      totalOperationsCleared: currentCleared,
      activePhase,
      isUniverseConcluded
    };
  }

  public async getAuthoritativeCampaignState(): Promise<UniverseCampaignState> {
    const data = await this.redisClient.hGetAll('wfm:campaign:global_state');
    if (!data || !data.activePhase) {
      return { currentGlobalProgressPercent: 0.0, totalOperationsCleared: 0, activePhase: 'INITIAL_CONTACT', isUniverseConcluded: false };
    }

    return {
      currentGlobalProgressPercent: parseFloat(data.currentGlobalProgressPercent),
      totalOperationsCleared: parseInt(data.totalOperationsCleared),
      activePhase: data.activePhase as any,
      isUniverseConcluded: data.isUniverseConcluded === 'true'
    };
  }
}

-- scripts/migrations/06_global_campaign_timeline.sql

CREATE TABLE IF NOT EXISTS campaign_global_metadata (
    id INT PRIMARY KEY DEFAULT 1,
    progress_percent NUMERIC(5,2) DEFAULT 0.00,
    current_phase VARCHAR(32) DEFAULT 'INITIAL_CONTACT',
    is_concluded BOOLEAN DEFAULT FALSE,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT single_row_restriction CHECK (id = 1) -- Enforces a single master timeline row record
);

-- Seed initial universal timeline state on deployment initialization

3. Real-Time Timeline Progression Route (artifacts/api-server/src/routes/campaign-timeline.ts)

This endpoint acts as the central check gate for the Hangar Deck interface. It provides real-time information to connecting pilots, showing the exact global campaign percentage and locking out expired operations automatically.
typescript
// artifacts/api-server/src/routes/campaign-timeline.ts
import { Router, Request, Response } from 'express';
import { GlobalCampaignDirector } from '../game/global-campaign-director';
import { Pool } from 'pg';

export const timelineRouter = Router();
const dbPool = new Pool({ connectionString: process.env.DATABASE_URL });
const campaignDirector = new GlobalCampaignDirector(
  `redis://${process.env.REDIS_STAGING_HOST || '127.0.0.1'}:${process.env.REDIS_STAGING_PORT || 6379}`,
  dbPool
);

campaignDirector.initCampaignDirector();

/**
 * Global Storyline Progress Tracking Route
 * Returns the exact server-authoritative completion percentage to connecting clients.
 */
timelineRouter.get('/current-timeline-state', async (req: Request, res: Response): Promise<void> => {
  try {
    const campaignState = await campaignDirector.getAuthoritativeCampaignState();

    res.status(200).json({
      success: true,
      globalProgressPercent: campaignState.currentGlobalProgressPercent,
      activePhase: campaignState.activePhase,
      isUniverseConcluded: campaignState.isUniverseConcluded,
      unlockedOperations: {
        contact: true, // Contact operation is always open for training verification loops
        beacon: campaignState.currentGlobalProgressPercent >= 25.0,
        warden: campaignState.currentGlobalProgressPercent >= 50.0,
        cage: campaignState.currentGlobalProgressPercent >= 75.0
      }
    });
  } catch (error) {
    res.status(500).json({ error: "HTTP 500: Global campaign timeline tracker failure." });
  }
});
