🕳️ 1. Server-Authoritative Gravity Well Engine (lib/gravity-simulation.ts)

This module simulates localized deep-space anomalies. It calculates continuous gravitational pulls based on proximity, applying vector forces directly within the physics tick to alter ship trajectories without breaking client-side parity.
typescript
// lib/gravity-simulation.ts
import { ENGINE_CONFIG, createDeterministicVector, Vector2D } from './deterministic-core';

export interface BlackHoleAnomaly {
  uid: string;
  globalX: number;
  globalY: number;
  eventHorizonRadius: number;
  gravitationalPullForce: number; // Force multiplier scale
}

export class GravitySimulationEngine {
  /**
   * Evaluates proximity vectors and applies authoritative gravitational pull forces.
   * Modifies velocity arrays deterministically using strict single-precision floats.
   */
  public applyAnomalyGravity(
    shipX: number,
    shipY: number,
    vx: number,
    vy: number,
    anomaly: BlackHoleAnomaly
  ): { nextVx: number; nextVy: number; insideHorizon: boolean } {
    const dx = Math.fround(anomaly.globalX - shipX);
    const dy = Math.fround(anomaly.globalY - shipY);
    const distanceSq = Math.fround(dx * dx + dy * dy);
    const distance = Math.fround(Math.sqrt(distanceSq));

    // 1. Check if the ship has crossed the fatal event horizon threshold
    if (distance <= Math.fround(anomaly.eventHorizonRadius)) {
      return { nextVx: 0.0, nextVy: 0.0, insideHorizon: true };
    }

    // 2. Apply inverse-square gravity pull decay within the tracking field boundary area
    const pullFieldRadius = Math.fround(anomaly.eventHorizonRadius * 12.0);
    if (distance < pullFieldRadius) {
      // Avoid division-by-zero errors at boundary pins
      const safeDistance = Math.max(1.0, distance);
      const pullIntensity = Math.fround(anomaly.gravitationalPullForce / (safeDistance * safeDistance));

      const dirX = Math.fround(dx / safeDistance);
      const dirY = Math.fround(dy / safeDistance);

      const nextVx = Math.fround(vx + Math.fround(dirX * pullIntensity * 8.5));
      const nextVy = Math.fround(vy + Math.fround(dirY * pullIntensity * 8.5));

      return { nextVx, nextVy, insideHorizon: false };
    }

    return { nextVx: vx, nextVy: vy, insideHorizon: false };
  }
}

💱 2. Space Station Trading Interface Node (components/TradingStationNode.tsx)

This interface component connects to your profile database layer. It lets pilots sell procedural iridium minerals harvested during exploration missions to replenish their credit fuel balances cleanly.
tsx
// artifacts/war-for-mars/src/game/TradingStationNode.tsx
import React, { useState } from 'react';

interface TradingNodeProps {
  initialCredits: number;
  initialIridium: number;
  onTransactionComplete: (newCredits: number, newIridium: number) => void;
}

export const TradingStationNode: React.FC<TradingNodeProps> = ({
  initialCredits,
  initialIridium,
  onTransactionComplete
}) => {
  const [credits, setCredits] = useState(initialCredits);
  const [iridium, setIridium] = useState(initialIridium);
  const [tradeAmount, setTradeAmount] = useState(10);

  const IRIDIUM_CREDIT_VALUE = 4; // Exchange rate matrix value: 1 Iridium unit = 4 Credits

  const executeSellTransaction = () => {
    if (iridium >= tradeAmount) {
      const creditGain = tradeAmount * IRIDIUM_CREDIT_VALUE;
      const nextCredits = credits + creditGain;
      const nextIridium = iridium - tradeAmount;

      setCredits(nextCredits);
      setIridium(nextIridium);
      onTransactionComplete(nextCredits, nextIridium);
      console.log(`[Market Success] Exchanged ${tradeAmount} Iridium for ${creditGain} server credits.`);
    } else {
      alert("MARKET ERROR: Insufficient mineral cargo available in vessel hull holding vectors.");
    }
  };

  return (
    <div className="w-full max-w-md bg-slate-900 border border-slate-800 p-4 rounded-xl font-mono text-xs text-slate-200 shadow-2xl mt-4">
      <div className="flex justify-between items-center border-b border-slate-800 pb-2 mb-3">
        <span className="text-emerald-400 font-bold uppercase">🛰️ ORBITAL REFINERY MARKET COMMERCE</span>
        <span className="bg-slate-950 border border-slate-800 px-2 py-0.5 rounded text-[10px] text-slate-400">
          STATION CHANNEL SECURE
        </span>
      </div>

      <div className="grid grid-cols-2 gap-2 mb-4 bg-slate-950 p-2 rounded border border-slate-800">
        <div>CARGO CAPACITY: <span className="text-amber-400 font-bold">{iridium} IRIDIUM</span></div>
        <div>LIQUID BALANCE: <span className="text-cyan-400 font-bold">{credits} CREDITS</span></div>
      </div>

      <div className="space-y-3">
        <div>
          <label className="block text-[10px] text-slate-400 uppercase mb-1">Exchange Cargo Volume Units</label>
          <input
            type="number"
            value={tradeAmount}
            onChange={(e) => setTradeAmount(Math.max(1, parseInt(e.target.value) || 0))}
            className="w-full bg-slate-950 border border-slate-800 p-2 rounded text-slate-100 font-bold"
            min="1"
            max={iridium}
          />
        </div>

        <div className="text-[10px] text-slate-500 bg-slate-950/40 p-1.5 rounded border border-slate-900/60">
          Current Exchange Protocol: 1.0 Unit Raw Iridium ──> Yields {IRIDIUM_CREDIT_VALUE}.0 Liquid Credits
        </div>

        <button
          onClick={executeSellTransaction}
          disabled={iridium <= 0}
          className={`w-full py-2.5 rounded font-bold uppercase transition-all tracking-wider text-center border ${
            iridium > 0
              ? 'bg-emerald-950 border-emerald-600 text-emerald-400 hover:bg-emerald-900/60'
              : 'bg-slate-950 border-slate-800 text-slate-600 cursor-not-allowed'
          }`}
        >
          💰 Transmit Cargo to Station Ledger
        </button>
      </div>
    </div>
  );
};
