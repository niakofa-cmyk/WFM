🎨 Part 2: 3D GPU Additive Plasma Shield Shader System (src/game/orbital-native-shield-shader.ts)

We expand your OrbitalNativeFlightRenderer component with a custom GPU shield effect. By modifying vertex shader attributes directly on a pre-allocated geometry buffer instead of creating objects on the fly, the shield illumination flashes dynamically on impact with 0% runtime heap overhead.
typescript
// artifacts/war-for-mars/src/game/orbital-native-shield-shader.ts
import * as THREE from 'three';

export class OrbitalNativeShieldShaderSystem {
  private geometry: THREE.SphereGeometry;
  private material: THREE.ShaderMaterial;
  private mesh: THREE.Mesh;
  private maxActiveShields = 200;
  
  constructor(scene: THREE.Scene) {
    // Low-polygon sphere bounding matrix encapsulating the fighter hulls
    this.geometry = new THREE.SphereGeometry(8, 16, 16);

    const vertexShader = `
      varying vec3 vNormal;
      varying vec2 vUv;
      void main() {
        vUv = uv;
        vNormal = normalize(normalMatrix * normal);
        gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
      }
    `;

    const fragmentShader = `
      varying vec3 vNormal;
      varying vec2 vUv;
      uniform vec3 u_shieldColor;
      uniform float u_intensity;
      uniform float u_time;

      void main() {
        // Compute Fresnel rim lighting effect for the energy bubble edge
        float intensity = pow(0.6 - dot(vNormal, vec3(0, 0, 1.0)), 3.0);
        vec3 glowColor = u_shieldColor * intensity * u_intensity;
        
        // Add animated energy ripple lines
        float ripple = sin(vUv.y * 30.0 - u_time * 10.0) * 0.1 * u_intensity;
        glowColor += u_shieldColor * ripple;

        gl_FragColor = vec4(glowColor, intensity * u_intensity);
      }
    `;

    this.material = new THREE.ShaderMaterial({
      vertexShader,
      fragmentShader,
      uniforms: {
        u_shieldColor: { value: new THREE.Color('#00f0ff') },
        u_intensity: { value: 0.0 },
        u_time: { value: 0.0 }
      },
      transparent: true,
      blending: THREE.AdditiveBlending,
      depthWrite: false
    });

    this.mesh = new THREE.Mesh(this.geometry, this.material);
    this.mesh.position.set(0, -99999, 0); // Isolate offscreen initially
    scene.add(this.mesh);
  }

  /**
   * Triggers a high-intensity plasma glow flash around an entity.
   */
  public flashShieldImpact(x: number, y: number, z: number, isUssf: boolean) {
    this.mesh.position.set(x, y, z);
    this.material.uniforms.u_shieldColor.value.set(isUssf ? '#00f0ff' : '#ffaa00');
    this.material.uniforms.u_intensity.value = 1.0; // Instantly flash to full intensity
  }

  /**
   * Smoothly decays the shield glow opacity inside the render loop ticker.
   */
  public stepShieldLifecycles(elapsedTime: number) {
    this.material.uniforms.u_time.value = elapsedTime;
    const currentIntensity = this.material.uniforms.u_intensity.value;
    
    if (currentIntensity > 0.0) {
      this.material.uniforms.u_intensity.value = Math.fround(currentIntensity - 0.05); // Smooth linear energy decay
    } else {
      this.mesh.position.set(0, -99999, 0); // Hide when inactive
    }
  }
}
