The root cause of the persistent 264,322 B gzip bundle audit failure within the standalone Combat Lab compilation is a static entrypoint import leak. To resolve this issue, you must implement a strict architectural sandbox rule: isolate all structural game logic data fields from the visual canvas components. 2. Fully Sandboxed Asynchronous Dynamic Viewport Chunk (src/game/Orbital3DCanvas.tsx)

Isolate all Three.js calls entirely within a single, autonomous component block. The main bundle must interact with this file only via plain serialized data payloads.
tsx
// artifacts/war-for-mars/src/game/Orbital3DCanvas.tsx
import React, { useEffect, useRef } from 'react';
import * as THREE from 'three';
import { RenderingPayload } from './types';

export const Orbital3DCanvas: React.FC<{ payload: RenderingPayload }> = ({ payload }) => {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const sceneRef = useRef<THREE.Scene | null>(null);
  const rendererRef = useRef<THREE.WebGLRenderer | null>(null);
  const instancedMeshRef = useRef<THREE.InstancedMesh | null>(null);

  useEffect(() => {
    if (!containerRef.current) return;

    // 1. Hardware Accelerated Render Pipeline Set-Up
    const scene = new THREE.Scene();
    sceneRef.current = scene;
    scene.background = new THREE.Color('#010104');

    const camera = new THREE.PerspectiveCamera(60, 800 / 500, 0.1, 50000);
    const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
    rendererRef.current = renderer;
    renderer.setSize(800, 500);
    containerRef.current.appendChild(renderer.domElement);

    // 2. Ultra-Lightweight Instanced Allocation Array
    const geom = new THREE.ConeGeometry(3, 10, 4);
    const mat = new THREE.MeshBasicMaterial({ color: '#ff3b30' });
    const mesh = new THREE.InstancedMesh(geom, mat, 1000);
    instancedMeshRef.current = mesh;
    scene.add(mesh);

    camera.position.set(0, 400, 100);
    camera.lookAt(0, 0, 0);

    let active = true;
    const animate = () => {
      if (!active) return;
      renderer.render(scene, camera);
      requestAnimationFrame(animate);
    };
    animate();

    return () => {
      active = false;
      geom.dispose();
      mat.dispose();
      renderer.dispose();
      if (containerRef.current && renderer.domElement) {
        containerRef.current.removeChild(renderer.domElement);
      }
    };
  }, []);

  // 3. React-to-WebGL Context Mutation Bridge
  useEffect(() => {
    const mesh = instancedMeshRef.current;
    if (!mesh) return;

    const dummy = new THREE.Object3D();
    payload.traffic.forEach((ship, idx) => {
      // Translate global coordinates relative to player center camera focus points
      dummy.position.set(ship.x - payload.playerX, 0, ship.y - payload.playerY);
      dummy.rotation.y = ship.heading * (Math.PI / 180);
      dummy.updateMatrix();
      mesh.setMatrixAt(idx, dummy.matrix);
    });
    mesh.instanceMatrix.needsUpdate = true;
  }, [payload]);

  return <div ref={containerRef} className="w-full h-[500px]" />;
};
export default Orbital3DCanvas;

3. Strict Bundler Optimization Splitting Layout (vite.config.ts)

Enforce absolute boundaries on Rollup. This forces Three.js out of the standard initialization asset blocks, permanently addressing the failing bundle budget gate.
typescript
// artifacts/war-for-mars/vite.config.ts
import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';

export default defineConfig({
  plugins: [react()],
  build: {
    target: 'esnext',
    chunkSizeWarningLimit: 140, // 140 KiB Target Audit Limit
    rollupOptions: {
      output: {
        manualChunks(id) {
          // Explicit chunk-splitting sandbox rules
          if (id.includes('node_modules/three')) {
            return 'three-core-engine-chunk';
          }
          if (id.includes('src/game/Orbital3DCanvas')) {
            return 'three-rendering-canvas-chunk';
          }
          if (id.includes('node_modules/react')) {
            return 'framework-react-vendor';
          }
        }
      }
    }
  }
});
