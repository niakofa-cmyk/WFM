🕳️ 1. 3D Space-Time Gravitational Distortion Shader Array (src/game/GravitationalDistortionShader.tsx)

This module wraps your viewport mesh inside a custom vertex-and-fragment post-processing pass (THREE.ShaderMaterial). As a pilot draws closer to a black hole coordinate, it maps an asymmetrical gravitational lensing ripple directly across the screen space on the GPU.
tsx
// artifacts/war-for-mars/src/game/GravitationalDistortionShader.tsx
import React, { useEffect, useRef } from 'react';
import * as THREE from 'three';

interface DistortionProps {
  proximityToHorizon: number; // Scale from 0.0 (Far) to 1.0 (Inside Horizon)
}

export const GravitationalDistortionShader: React.FC<DistortionProps> = ({ proximityToHorizon }) => {
  const containerRef = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    if (!containerRef.current) return;

    const width = 800;
    const height = 500;
    const scene = new THREE.Scene();
    const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);
    const renderer = new THREE.WebGLRenderer({ antialias: false });
    renderer.setSize(width, height);
    containerRef.current.appendChild(renderer.domElement);

    const geometry = new THREE.PlaneGeometry(2, 2);

    const vertexShader = `
      varying vec2 vUv;
      void main() {
        vUv = uv;
        gl_Position = vec4(position, 1.0);
      }
    `;

    const fragmentShader = `
      varying vec2 vUv;
      uniform float u_distortionStrength;
      uniform float u_time;

      void main() {
        vec2 uv = vUv;
        
        // Compute gravitational lensing displacement vector
        if (u_distortionStrength > 0.0) {
          vec2 centerDist = uv - vec2(0.5, 0.5);
          float radius = length(centerDist);
          
          // Asymmetrical spacetime distortion wave
          float distortion = sin(radius * 20.0 - u_time * 4.0) * 0.02 * u_distortionStrength;
          uv += (centerDist / (radius + 0.01)) * distortion;
        }

        // Render deep space backdrop gradient pass
        vec3 color = vec3(0.02, 0.02, 0.05);
        
        // Add singularity core darkening
        float centerGlow = length(uv - vec2(0.5, 0.5));
        color *= smoothstep(0.08 * u_distortionStrength, 0.35, centerGlow);

        gl_FragColor = vec4(color, 1.0);
      }
    `;

    const uniforms = {
      u_distortionStrength: { value: Math.fround(proximityToHorizon) },
      u_time: { value: 0.0 }
    };

    const material = new THREE.ShaderMaterial({
      vertexShader,
      fragmentShader,
      uniforms,
      depthWrite: false
    });

    const mesh = new THREE.Mesh(geometry, material);
    scene.add(mesh);

    let active = true;
    const clock = new THREE.Clock();

    const render = () => {
      if (!active) return;
      uniforms.u_distortionStrength.value = Math.fround(proximityToHorizon);
      uniforms.u_time.value = clock.getElapsedTime();
      renderer.render(scene, camera);
      requestAnimationFrame(render);
    };
    render();

    return () => {
      active = false;
      geometry.dispose();
      material.dispose();
      renderer.dispose();
      if (containerRef.current && renderer.domElement) {
        containerRef.current.removeChild(renderer.domElement);
      }
    };
  }, [proximityToHorizon]);

  return <div ref={containerRef} className="w-full h-[500px] border border-red-950 rounded-lg overflow-hidden shadow-2xl shadow-red-950/10" />;
};

📊 2. Faction Cargo Board System Matrix (lib/faction-cargo-board.ts)

This tracking module reads pilot statistics from the server and provides trade contract payouts. It maps reputation data structures and scales credit rewards cleanly.
typescript
// lib/faction-cargo-board.ts
import { ENGINE_CONFIG } from './deterministic-core';

export interface CargoContract {
  contractId: string;
  factionRequirement: 'USSF' | 'ACCORD';
  requiredIridiumVolume: number;
  baseRewardCredits: number;
  completed: boolean;
}

export class FactionCargoBoardSystem {
  private activeContracts: CargoContract[] = [];

  constructor() {
    this.seedStagingContracts();
  }

  private seedStagingContracts() {
    this.activeContracts.push({
      contractId: 'con-ussf-mars-vanguard',
      factionRequirement: 'USSF',
      requiredIridiumVolume: 80,
      baseRewardCredits: 350,
      completed: false
    });

    this.activeContracts.push({
      contractId: 'con-accord-phobos-supply',
      factionRequirement: 'ACCORD',
      requiredIridiumVolume: 120,
      baseRewardCredits: 550,
      completed: false
    });
  }

  /**
   * Processes contract payouts by checking pilot ledger alignments.
   * Gives a 25% reward multiplier bonus if the pilot's verified kill count exceeds 10 targets.
   */
  public evaluateContractFulfillment(
    contractId: string,
    currentFaction: 'USSF' | 'ACCORD',
    pilotKills: number,
    cargoIridium: number
  ): { success: boolean; creditPayout: number; deductedIridium: number } {
    const contract = this.activeContracts.find(c => c.contractId === contractId && !c.completed);
    
    if (!contract) {
      return { success: false, creditPayout: 0, deductedIridium: 0 };
    }

    if (currentFaction !== contract.factionRequirement) {
      console.warn('[Cargo System Alert] Pilot faction registration does not match contract specifications.');
      return { success: false, creditPayout: 0, deductedIridium: 0 };
    }

    if (cargoIridium < contract.requiredIridiumVolume) {
      return { success: false, creditPayout: 0, deductedIridium: 0 };
    }

    // Process deterministic payout calculations
    let finalReward = contract.baseRewardCredits;
    if (pilotKills >= 10) {
      finalReward = Math.floor(Math.fround(finalReward * 1.25)); // 25% Combat Veteran Bonus
    }

    contract.completed = true;

    return {
      success: true,
      creditPayout: finalReward,
      deductedIridium: contract.requiredIridiumVolume
    };
  }

  public getAvailableContracts(faction: 'USSF' | 'ACCORD'): CargoContract[] {
    return this.activeContracts.filter(c => c.factionRequirement === faction && !c.completed);
  }
}

🎨 3. Interactive Cargo Board Dashboard Control Overlay

Create this component wrapper interface to deploy cargo delivery logs alongside your procedural deep-space flight engine nodes.
tsx
// artifacts/war-for-mars/src/game/FactionCargoInterface.tsx
import React, { useState } from 'react';
import { FactionCargoBoardSystem, CargoContract } from '../lib/faction-cargo-board';

export const FactionCargoInterface: React.FC<{
  pilotFaction: 'USSF' | 'ACCORD';
  kills: number;
  onRewardClaimed: (creditsEarned: number, iridiumLost: number) => void;
}> = ({ pilotFaction, kills, onRewardClaimed }) => {
  const [boardEngine] = useState(new FactionCargoBoardSystem());
  const [available, setAvailable] = useState<CargoContract[]>(boardEngine.getAvailableContracts(pilotFaction));

  const processClaim = (id: string) => {
    // Simulate checking warehouse inventory sizes (mocked at 150 units for validation checks)
    const outcome = boardEngine.evaluateContractFulfillment(id, pilotFaction, kills, 150);
    
    if (outcome.success) {
      onRewardClaimed(outcome.creditPayout, outcome.deductedIridium);
      setAvailable(boardEngine.getAvailableContracts(pilotFaction));
      alert(`✓ CONTRACT SECURED: Disbursed ${outcome.creditPayout} credits to pilot ledger profiles.`);
    } else {
      alert("🔒 CLEARANCE DENIED: Requirements for cargo delivery nodes are incomplete.");
    }
  };

  return (
    <div className="w-full max-w-4xl bg-slate-900 border border-slate-800 p-4 rounded-xl mt-4 font-mono text-xs text-slate-300">
      <div className="text-xs uppercase font-bold tracking-widest text-slate-400 mb-3 text-red-500">
        🛰️ FACTION COMMODITY MANIFEST SYSTEM
      </div>
      
      {available.length === 0 ? (
        <div className="p-4 bg-slate-950 border border-slate-800 text-slate-500 text-center rounded-lg">
          No remaining operational cargo paths open for your faction profile rank index.
        </div>
      ) : (
        <div className="space-y-3">
          {available.map(c => (
            <div key={c.contractId} className="p-3 bg-slate-950 border border-slate-800 rounded-lg flex justify-between items-center">
              <div>
                <div className="font-bold text-slate-200 uppercase">{c.contractId.replace('con-', '').replace(/-/g, ' ')}</div>
                <div className="text-[10px] text-slate-400 mt-1">
                  MANDATORY CARGO: <span className="text-cyan-400">{c.requiredIridiumVolume} Units Iridium</span>
                </div>
              </div>
              <button
                onClick={() => processClaim(c.contractId)}
                className="px-4 py-2 bg-slate-900 border border-slate-700 hover:border-slate-500 text-slate-200 rounded font-bold transition-all uppercase"
              >
                🔒 Deliver Manifest [+{c.baseRewardCredits} Credits]
              </button>
            </div>
          ))}
        </div>
      )}
    </div>
  );
};
