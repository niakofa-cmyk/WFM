☢️ 1. Continuous Deep-Space Radiation Simulation (lib/radiation-hazard-engine.ts)

This module simulates environmental friction in uncharted dark space. It monitors the pilot's 64-bit coordinate displacement continuously, calculation scaling shield erosion metrics without relying on rigid regional zone triggers.
typescript
// lib/radiation-hazard-engine.ts
import { ENGINE_CONFIG } from './deterministic-core';
import { InfiniteCoordinates } from './spatial-origin-engine';

export interface RadiationStatus {
  sievertsPerSecond: number;
  shieldDegradationMultiplier: number;
  isHullBreached: boolean;
}

export class RadiationHazardEngine {
  // Colonized safety radius boundary index (Equivalent to 5 complete space sectors away from home base)
  private deepSpaceThresholdSectors = 5n; 

  /**
   * Evaluates absolute scalar coordinate offsets to determine environmental radiation density.
   * Enforces strict single-precision floating-point bounds during calculation steps.
   */
  public evaluateSpaceEnvironment(current: InfiniteCoordinates): RadiationStatus {
    const absSectorX = current.sectorX < 0n ? -current.sectorX : current.sectorX;
    const absSectorY = current.sectorY < 0n ? -current.sectorY : current.sectorY;
    
    // Determine gross taxicab distance block spans away from origin (0,0)
    const totalSectorDistance = absSectorX + absSectorY;

    if (totalSectorDistance <= this.deepSpaceThresholdSectors) {
      return { sievertsPerSecond: 0.0, shieldDegradationMultiplier: 1.0, isHullBreached: false };
    }

    // Scale radiation intensity exponentially based on deep space penetration levels
    const hazardFactor = Number(totalSectorDistance - this.deepSpaceThresholdSectors);
    const sievertsPerSecond = Math.fround(0.05 * hazardFactor + Math.fround(current.localX / ENGINE_CONFIG.CHUNK_SIZE) * 0.01);
    
    // Accelerate shield degradation properties linearly
    const shieldDegradationMultiplier = Math.fround(1.0 + Math.fround(0.12 * hazardFactor));
    const isHullBreached = sievertsPerSecond > 2.50; // Fatal dose warning threshold

    return {
      sievertsPerSecond,
      shieldDegradationMultiplier,
      isHullBreached
    };
  }
}

🧭 2. 3D Absolute Waypoint Compass Ring (src/game/orbital-waypoint-compass.ts)

This rendering component adds an arrow indicator mesh to the ship's model group. It calculates inverse spatial coordinate tangents to point directly toward the deployment origin, bypassing camera changes.
typescript
// artifacts/war-for-mars/src/game/orbital-waypoint-compass.ts
import * as THREE from 'three';
import { InfiniteCoordinates } from '../lib/spatial-origin-engine';
import { ENGINE_CONFIG } from '../lib/deterministic-core';

export class OrbitalNativeWaypointCompass {
  private compassGroup: THREE.Group;
  private arrowMesh: THREE.Mesh;
  private coreRingMesh: THREE.Mesh;

  constructor(parentShipGroup: THREE.Group) {
    this.compassGroup = new THREE.Group();
    this.compassGroup.position.set(0, 2, 0); // Elevate slightly above player hull plane

    // 1. Construct thin vector tracking track ring
    const ringGeom = new THREE.RingGeometry(8, 8.5, 32);
    ringGeom.rotateX(-Math.PI / 2);
    const ringMat = new THREE.MeshBasicMaterial({ color: '#1e293b', transparent: true, opacity: 0.6 });
    this.coreRingMesh = new THREE.Mesh(ringGeom, ringMat);
    this.compassGroup.add(this.coreRingMesh);

    // 2. Construct precise directional pointer arrow geometry mesh
    const arrowGeom = new THREE.ConeGeometry(1.5, 5, 3);
    arrowGeom.rotateX(Math.PI / 2); // Lay flat facing forward
    arrowGeom.translate(0, 0, -9.5); // Snap arrow to sit exactly on the outer ring edge border
    const arrowMat = new THREE.MeshBasicMaterial({ color: '#f59e0b', toneMapped: false });
    this.arrowMesh = new THREE.Mesh(arrowGeom, arrowMat);
    this.compassGroup.add(this.arrowMesh);

    parentShipGroup.add(this.compassGroup);
  }

  /**
   * Computes the line-of-sight vector back to the deployment coordinates.
   * Keeps arrow direction accurate regardless of player rotation or distance.
   */
  public updateCompassVector(current: InfiniteCoordinates, homeBase: InfiniteCoordinates, shipHeadingRad: number) {
    const sectorDx = homeBase.sectorX - current.sectorX;
    const sectorDy = homeBase.sectorY - current.sectorY;

    // Convert big integers back into scale metrics
    const globalDx = Math.fround(Number(sectorDx) * ENGINE_CONFIG.CHUNK_SIZE + (homeBase.localX - current.localX));
    const globalDy = Math.fround(Number(sectorDy) * ENGINE_CONFIG.CHUNK_SIZE + (homeBase.localY - current.localY));

    // Calculate absolute trigonometric angular direction back to home base station
    const targetAngleRad = Math.atan2(globalDx, globalDy);

    // Negate the ship's rotation matrix to maintain absolute compass pointing references
    this.arrowMesh.rotation.y = targetAngleRad - shipHeadingRad;

    // Shift ring color state to bright crimson if hazard values rise
    const distance = Math.sqrt(globalDx * globalDx + globalDy * globalDy);
    const mat = this.arrowMesh.material as THREE.MeshBasicMaterial;
    if (distance > 20000) {
      mat.color.set('#ef4444'); // Dangerous distance tracking indicator
    } else {
      mat.color.set('#f59e0b'); // Stable transit path
    }
  }

  public dispose() {
    this.coreRingMesh.geometry.dispose();
    (this.coreRingMesh.material as THREE.MeshBasicMaterial).dispose();
    this.arrowMesh.geometry.dispose();
    (this.arrowMesh.material as THREE.MeshBasicMaterial).dispose();
  }
}
