1. Hardware-Agnostic Infinite Space Manager (lib/spatial-origin-engine.ts)

This engine module replaces all camera snaps and edge redirects with an automated origin shifting process. When a player moves too far from the local tracking center point \((0,0)\), the system shifts the simulation zero-point directly to the player, updating a 64-bit integer matrix to keep track of their exact location.
typescript
// lib/spatial-origin-engine.ts
import { ENGINE_CONFIG, createDeterministicVector, Vector2D } from './deterministic-core';

export interface InfiniteCoordinates {
  sectorX: bigint; // Light-year scale positioning blocks
  sectorY: bigint;
  localX: number;  // Single-precision localized float metrics inside sector boundaries
  localY: number;
}

export class SpatialOriginEngine {
  // Boundary window threshold before executing an engine origin shift (4096 units width)
  private shiftThreshold = Math.fround(ENGINE_CONFIG.CHUNK_SIZE);

  /**
   * Updates localized coordinates continuously. 
   * If the ship drifts beyond the boundary threshold, the origin moves to protect precision bits.
   * This completely avoids camera snaps, letting players travel deep into deep space.
   */
  public updateContinuousFlight(current: InfiniteCoordinates, vx: number, vy: number): InfiniteCoordinates {
    // 1. Linearly apply single-precision velocity offsets across frame steps
    let nextLocalX = Math.fround(current.localX + vx);
    let nextLocalY = Math.fround(current.localY + vy);

    let nextSectorX = current.sectorX;
    let nextSectorY = current.sectorY;

    // 2. Continuous Sector Boundary Evaluation Matrix
    if (nextLocalX > this.shiftThreshold) {
      nextSectorX += 1n;
      nextLocalX = Math.fround(nextLocalX - this.shiftThreshold);
    } else if (nextLocalX < -this.shiftThreshold) {
      nextSectorX -= 1n;
      nextLocalX = Math.fround(nextLocalX + this.shiftThreshold);
    }

    if (nextLocalY > this.shiftThreshold) {
      nextSectorY += 1n;
      nextLocalY = Math.fround(nextLocalY - this.shiftThreshold);
    } else if (nextLocalY < -this.shiftThreshold) {
      nextSectorY -= 1n;
      nextLocalY = Math.fround(nextLocalY + this.shiftThreshold);
    }

    return {
      sectorX: nextSectorX,
      sectorY: nextSectorY,
      localX: nextLocalX,
      localY: nextLocalY
    };
  }

  /**
   * Calculates the exact Euclidean distance a pilot must fly to return to a base point.
   */
  public calculateReturnRange(current: InfiniteCoordinates, target: InfiniteCoordinates): number {
    const sectorDx = current.sectorX - target.sectorX;
    const sectorDy = current.sectorY - target.sectorY;

    // Map big integers back into scale metrics
    const totalGlobalDx = Math.fround(Number(sectorDx) * ENGINE_CONFIG.CHUNK_SIZE + (current.localX - target.localX));
    const totalGlobalDy = Math.fround(Number(sectorDy) * ENGINE_CONFIG.CHUNK_SIZE + (current.localY - target.localY));

    return Math.fround(Math.sqrt(Math.fround(totalGlobalDx * totalGlobalDx + totalGlobalDy * totalGlobalDy)));
  }
}

3. Continuous 3D Spatial Viewport Hub Component (components/ContinuousFlightUniverse.tsx)

This updated UI workspace links your camera matrices directly to the pilot's localized position, removing all camera overrides or snappy edge boundaries to deliver an unthrottled, open universe flight feel.
tsx
// artifacts/war-for-mars/src/game/ContinuousFlightUniverse.tsx
import React, { useEffect, useRef, useState } from 'react';
import * as THREE from 'three';
import { SpatialOriginEngine, InfiniteCoordinates } from '../lib/spatial-origin-engine';
import { SpatialTelemetryHUD } from './SpatialTelemetryHUD';

export const ContinuousFlightUniverse: React.FC = () => {
  const mountRef = useRef<HTMLDivElement | null>(null);
  const [distanceToTarget, setDistanceToTarget] = useState<number>(0);
  const [speedMetrics, setSpeedMetrics] = useState<number>(0);

  const spatialEngine = useRef(new SpatialOriginEngine());
  
  // Lock starting anchor point coordinates to verify return flight metrics
  const deploymentOrigin = useRef<InfiniteCoordinates>({ sectorX: 0n, sectorY: 0n, localX: 0, localY: 0 });
  const currentFlightCoords = useRef<InfiniteCoordinates>({ sectorX: 0n, sectorY: 0n, localX: 0, localY: 0 });
  
  const inputState = useRef({ thrust: false, left: false, right: false });
  const physicsVelocity = useRef({ x: 0, y: 0, heading: 0 });

  useEffect(() => {
    if (!mountRef.current) return;

    const width = mountRef.current.clientWidth || 800;
    const height = 500;

    const scene = new THREE.Scene();
    scene.background = new THREE.Color('#010103');
    
    // Add grid layout markers to visualize endless cosmic locomotion loops
    const gridHelper = new THREE.GridHelper(8192, 64, '#1e293b', '#0f172a');
    scene.add(gridHelper);

    const camera = new THREE.PerspectiveCamera(60, width / height, 0.1, 20000);
    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(width, height);
    mountRef.current.appendChild(renderer.domElement);

    // Deploy Player Space Vessel Mesh
    const shipGeom = new THREE.ConeGeometry(4, 12, 4);
    shipGeom.rotateX(Math.PI / 2);
    const shipMat = new THREE.MeshBasicMaterial({ color: '#ef4444', wireframe: true });
    const shipMesh = new THREE.Mesh(shipGeom, shipMat);
    scene.add(shipMesh);

    // Keyboard capture controls
    const handleKeyEvent = (e: KeyboardEvent, isDown: boolean) => {
      if (e.key === 'w' || e.key === 'ArrowUp')    inputState.current.thrust = isDown;
      if (e.key === 'a' || e.key === 'ArrowLeft')  inputState.current.left = isDown;
      if (e.key === 'd' || e.key === 'ArrowRight') inputState.current.right = isDown;
    };
    window.addEventListener('keydown', (e) => handleKeyEvent(e, true));
    window.addEventListener('keyup', (e) => handleKeyEvent(e, false));

    let frameId: number;
    let lastTimestamp = performance.now();

    const loop = (now: number) => {
      const dt = 16; // Fix updates to strict 16ms ticks
      lastTimestamp = now;

      // 1. Process steering thrust velocity additions
      if (inputState.current.left)  physicsVelocity.current.heading -= 3.5;
      if (inputState.current.right) physicsVelocity.current.heading += 3.5;
      physicsVelocity.current.heading %= 360;

      const headingRad = physicsVelocity.current.heading * (Math.PI / 180);
      
      if (inputState.current.thrust) {
        physicsVelocity.current.x = Math.fround(physicsVelocity.current.x + Math.sin(headingRad) * 0.25);
        physicsVelocity.current.y = Math.fround(physicsVelocity.current.y + Math.cos(headingRad) * 0.25);
      }

      // Interstellar drag matrix dampening parameters
      physicsVelocity.current.x *= 0.985;
      physicsVelocity.current.y *= 0.985;

      // 2. Update continuous coordinate positions without resetting camera values
      currentFlightCoords.current = spatialEngine.current.updateContinuousFlight(
        currentFlightCoords.current,
        physicsVelocity.current.x,
        physicsVelocity.current.y
      );

      // 3. Keep the tracking camera firmly pinned directly above the moving ship asset
      const sx = currentFlightCoords.current.localX;
      const sz = currentFlightCoords.current.localY;
      
      shipMesh.position.set(sx, 0, sz);
      shipMesh.rotation.y = headingRad;

      camera.position.set(sx, 450, sz + 150); // Orthogonal structural tracking focus
      camera.lookAt(sx, 0, sz);

      // Shift environmental grids dynamically beneath the player to simulate continuous motion
      gridHelper.position.set(
        -Math.fround(Number(currentFlightCoords.current.sectorX) * 4096) % 8192,
        -1,
        -Math.fround(Number(currentFlightCoords.current.sectorY) * 4096) % 8192
      );

      // 4. Push live telemetry state metrics out to HUD elements
      const currentDistance = spatialEngine.current.calculateReturnRange(currentFlightCoords.current, deploymentOrigin.current);
      const currentVelocityMag = Math.sqrt(physicsVelocity.current.x * physicsVelocity.current.x + physicsVelocity.current.y * physicsVelocity.current.y);

      setDistanceToTarget(currentDistance);
      setSpeedMetrics(currentVelocityMag);

      renderer.render(scene, camera);
      frameId = requestAnimationFrame(loop);
    };

    frameId = requestAnimationFrame(loop);

    return () => {
      cancelAnimationFrame(frameId);
      window.removeEventListener('keydown', (e) => handleKeyEvent(e, true));
      window.removeEventListener('keyup', (e) => handleKeyEvent(e, false));
      shipGeom.dispose();
      shipMat.dispose();
      gridHelper.dispose();
      renderer.dispose();
      if (mountRef.current && renderer.domElement) {
        mountRef.current.removeChild(renderer.domElement);
      }
    };
  }, []);

  return (
    <div className="relative w-full h-[550px] bg-black border border-slate-800 rounded-xl overflow-hidden shadow-2xl">
      <div ref={mountRef} className="w-full h-full" />
      <SpatialTelemetryHUD 
        currentCoords={currentFlightCoords.current} 
        destinationCoords={deploymentOrigin.current} 
        distanceToTarget={distanceToTarget} 
        velocityMagnitude={speedMetrics} 
      />
    </div>
  );
};
